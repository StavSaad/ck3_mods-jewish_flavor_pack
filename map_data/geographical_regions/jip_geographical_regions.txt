# The mod's regions. Decisions name regions, not titles, wherever a place is meant, so a map mod's compatch replaces
# this file (and the landed titles) rather than the decisions. Duchy and kingdom lists follow a map's own de jure land;
# county lists and the cities' provinces may need the compatch's versions. A region nested in another must not share
# provinces with its siblings: the parent then counts them twice

# Israel within Canaan: Palestine, the Jordan and the Negev (Kerak, Moab, is Edom's)
jip_region_israel = {
	duchies = {
		d_palestine d_urdunn
	}
	counties = {
		c_negev
	}
}

# The Land of Israel of Restore Israel and Form Hebrew Culture: Palestine, the Jordan and Oultrejourdain
jip_region_land_of_israel = {

	}
}

# Canaan is mostly inside Beth Nahrain already, so only its Edomite and Arabian counties are added separately
# (listing jip_region_canaan here counted those provinces twice)

### The empires' claims: a kingdom whose de jure capital lies in the region joins the empire

# Canaan: the kingdoms its founder holds. Israel's, Phoenicia's and Edom's capitals (Jerusalem, Beirut, Kerak) lie in
# these two, so jip_region_canaan isn't added (it would list their provinces twice)
jip_region_canaan_claims = {
	kingdoms = {
		k_syria k_jerusalem
	}
}

# Canaan: the islands, if its founder also completely controls them
jip_region_canaan_isles = {
	kingdoms = {
		k_cyprus
	}
}

# Beth Nahrain (beyond the Jazira and Mesopotamia, which it needs): the kingdoms its founder holds and completely controls
jip_region_beth_nahrain_claims = {
	kingdoms = {
		k_syria k_jerusalem
	}
}

# Aram Rabba (beyond Syria, which it takes by name): the kingdoms its founder completely controls, Zenobia's
# conquests among them. Not Syria itself, which would be taken twice; Phoenicia's coast brings in the kingdoms carved
# from it
jip_region_aram_claims = {
	regions = {
		jip_region_phoenicia
	}
	kingdoms = {
		k_jazira k_jerusalem k_arabia k_egypt
	}
}
