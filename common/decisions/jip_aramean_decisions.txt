# Arameanism (0.15 plan, section 14)

# Restore the Gods of Aram: an independent ruler of Syriac heritage (Aramean, Assyrian, Mandaean) who holds and controls the counties of
# Damascus, Palmyra and Manbij (jip_city_<name> regions), the cities of the three great temples, turns back to Hadad
# (jip_restore_gods_of_aram_effect)
jip_restore_gods_of_aram_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds"
	}
	desc = jip_restore_gods_of_aram_decision_desc
	selection_tooltip = jip_restore_gods_of_aram_decision_tooltip
	decision_group_type = major
	ai_check_interval = 60

	is_shown = {
		is_ruler = yes
		culture = { has_cultural_pillar = heritage_syriac }
		NOT = { faith = faith:aramean_pagan }
		NOT = { has_character_flag = jip_restored_gods_of_aram }
		OR = {
			jip_holds_city_county_trigger = { CITY = jip_city_damascus }
			jip_holds_city_county_trigger = { CITY = jip_city_palmyra }
			jip_holds_city_county_trigger = { CITY = jip_city_manbij }
		}
	}

	is_valid = {
		is_independent_ruler = yes
		prestige_level >= 2
		custom_tooltip = {
			text = jip_restore_gods_of_aram_cities_tt
			jip_city_county_in_realm_trigger = { CITY = jip_city_damascus }
			jip_city_county_in_realm_trigger = { CITY = jip_city_palmyra }
			jip_city_county_in_realm_trigger = { CITY = jip_city_manbij }
		}
		custom_tooltip = {
			text = jip_restore_gods_of_aram_control_tt
			jip_completely_controls_city_county_trigger = { CITY = jip_city_damascus }
			jip_completely_controls_city_county_trigger = { CITY = jip_city_palmyra }
			jip_completely_controls_city_county_trigger = { CITY = jip_city_manbij }
		}
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		is_at_war = no
	}

	cost = {
		prestige = 500
	}

	effect = {
		add_character_flag = jip_restored_gods_of_aram
		if = {
			limit = { jip_capital_is_city_trigger = { CITY = jip_city_palmyra } }
			jip_restore_gods_of_aram_effect = { RITE = belism }
		}
		else = {
			jip_restore_gods_of_aram_effect = { RITE = hadadism }
		}
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 0
		modifier = {
			add = 10
			has_trait = cynical
		}
	}
}
