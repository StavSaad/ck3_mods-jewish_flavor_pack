claim_exilarchate_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	ai_check_interval = 60
	desc = claim_exilarchate_decision_desc
	decision_group_type = major

	is_shown = {
		culture = { has_cultural_pillar = heritage_israelite }
		trigger_if = {
			limit = {
				has_faith = faith:rabbinic_faith
			}
			NOT = {
				has_claim_on = title:d_exilarchate
			}
			NOT = {
				has_title = title:d_exilarchate
			}
		}
		trigger_else_if = {
			limit = {
				has_faith = faith:karaism
			}
			NOT = {
				has_claim_on = title:d_karaite_exilarchate
			}
			NOT = {
				has_title = title:d_karaite_exilarchate
			}
		}
		trigger_else = {
			always = no
		}
	}

	is_valid = {
		is_independent_ruler = yes 		
		culture = { has_cultural_pillar = heritage_israelite }
		custom_tooltip = {
			text = no_active_claim_exilarchate_scheme
			NOT = {
				any_scheme = {
					type = claim_exilarchate
				}
			}
		}
	}

	effect = {
		if = {
			limit = {
				has_faith = faith:rabbinic_faith
			}
			title:d_exilarchate = {
				save_scope_as = target_title
			}
			begin_scheme_basic_effect = { 
				SCHEME_TYPE = claim_exilarchate 
				TARGET_TYPE = target_title 
				TARGET_SCOPE = title:d_exilarchate
			}
			trigger_event = { id = jip_claim_exilarchate.003 }
		}
		else_if = {
			limit = {
				has_faith = faith:karaism
			}
			title:d_karaite_exilarchate = {
				save_scope_as = target_title
			}
			begin_scheme_basic_effect = { 
				SCHEME_TYPE = claim_exilarchate 
				TARGET_TYPE = target_title 
				TARGET_SCOPE = title:d_karaite_exilarchate
			}
			trigger_event = { id = jip_claim_exilarchate.003 }
		}
		else = {
			# nothing
		}
	}

	cost = { prestige = { value = 500 } }

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

# A Christian or Muslim ruler expels the Jews from their realm, seizing half of every Jew's gold.
# The edict is stored on the ruler's primary title and bars Jewish families from settling there.
jip_expel_the_jews_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	ai_check_interval = 120
	desc = jip_expel_the_jews_decision_desc
	selection_tooltip = jip_expel_the_jews_decision_tooltip
	decision_group_type = major

	is_shown = {
		is_independent_ruler = yes
		is_landed = yes
		OR = {
			religion = religion:christianity_religion
			religion = religion:islam_religion
		}
		primary_title = {
			NOT = { has_variable = jip_jews_expelled }
		}
	}

	is_valid = {
		custom_tooltip = {
			text = jip_jews_edict_cooldown_tt
			primary_title = {
				NOT = { has_variable = jip_jews_edict_cooldown }
			}
		}
		custom_tooltip = {
			text = jip_expel_the_jews_no_families_tt
			OR = {
				any_vassal_or_below = {
					religion = religion:judaism_religion
				}
				# Families living in the realm without being its vassals
				any_ruler = {
					jip_mishpacha_in_realm_of_trigger = { RULER = root }
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
	}

	effect = {
		jip_expel_the_jews_effect = yes
	}

	ai_potential = {
		OR = {
			religion = religion:christianity_religion
			religion = religion:islam_religion
		}
	}

	# Weighed by personality, debts, and the tenets of their rite and personal faith
	ai_will_do = {
		base = 0
		jip_expel_the_jews_ai_modifier = yes
	}
}
