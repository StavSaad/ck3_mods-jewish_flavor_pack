# Character converts faith through the convert window or the matching script effect
# root = the converted character
on_faith_conversion = {
	on_actions = {
		# Delayed so vanilla's faith_conversion.0003 has converted the family first, keeping them out of the heir pick
		delay = { days = 1 }
		mishpacha_left_judaism_on_action
	}
}

# Yearly safety net for conversions that bypass on_faith_conversion, e.g. set_character_faith in events
# root = the character
yearly_playable_pulse = {
	on_actions = {
		mishpacha_left_judaism_on_action
		jip_mishpacha_ai_migration_on_action
		jip_mishpacha_succession_law_on_action
		jip_mishpacha_replace_estate_buildings_on_action
		jip_mishpacha_villa_on_action
		jip_mishpacha_contracts_on_action
	}
}

mishpacha_left_judaism_on_action = {
	trigger = {
		has_government = mishpacha_government
		NOT = { religion = religion:judaism_religion }
	}

	effect = {
		jip_mishpacha_leave_faith_effect = yes
	}
}

# An expelled family reaches its refuge
# root = the family head
jip_expulsion_refuge_arrival_on_action = {
	trigger = {
		has_variable = jip_expulsion_refuge
	}
	effect = {
		jip_mishpacha_settle_refuge_effect = yes
	}
}

# A fleeing family's journey was cut short (captured, killed, cancelled): their estate is still moved out
# root = the travelling character
on_travel_plan_abort = {
	on_actions = {
		jip_expulsion_refuge_arrival_on_action
	}
}

# AI families now and then weigh moving their estate to a more welcoming realm (about every three years)
# root = the family head
jip_mishpacha_ai_migration_on_action = {
	trigger = {
		jip_mishpacha_ai_can_migrate_trigger = yes
	}
	effect = {
		random = {
			chance = 33
			jip_mishpacha_ai_migration_effect = yes
		}
	}
}

# Safety net: a Mishpacha family head without a realm succession law can't have a succession built
# root = the character
jip_mishpacha_succession_law_on_action = {
	trigger = {
		has_government = mishpacha_government
		NOT = { has_realm_law = mishpacha_succession_law }
	}
	effect = {
		jip_mishpacha_ensure_succession_law_effect = yes
	}
}

# Swaps vanilla estate buildings hidden from Mishpacha families for their replacements. Runs yearly, and a day after game start,
# once vanilla has filled the estates (on_game_start_after_lobby)
# root = the family head
jip_mishpacha_replace_estate_buildings_on_action = {
	trigger = {
		has_government = mishpacha_government
		domicile ?= { jip_domicile_has_vanilla_influence_building_trigger = yes }
	}
	effect = {
		jip_mishpacha_replace_estate_buildings_effect = yes
	}
}

on_game_start_after_lobby = {
	on_actions = {
		jip_mishpacha_game_start_after_lobby_on_action
	}
}

jip_mishpacha_game_start_after_lobby_on_action = {
	effect = {
		every_ruler = {
			limit = { has_government = mishpacha_government }
			trigger_event = {
				on_action = jip_mishpacha_replace_estate_buildings_on_action
				days = 1
			}
			# The Villa exists from the start, so its prestige needs no delay
			jip_mishpacha_refresh_villa_modifier_effect = yes
			trigger_event = { on_action = jip_mishpacha_game_start_courtiers_on_action }
			trigger_event = {
				on_action = jip_mishpacha_contracts_new_area_on_action
				days = 1
			}
		}
	}
}

# Keeps the Villa prestige modifier in step with the Villa tier (jip_mishpacha_refresh_villa_modifier_effect).
# Runs yearly, at game start, when an estate building is completed, and a day after a title is gained
# (jip_title_on_actions.txt), so an heir gets it once the estate has passed to them.
# root = the character
jip_mishpacha_villa_on_action = {
	trigger = {
		OR = {
			has_government = mishpacha_government
			has_character_modifier = jip_mishpacha_villa_1
			has_character_modifier = jip_mishpacha_villa_2
			has_character_modifier = jip_mishpacha_villa_3
			has_character_modifier = jip_mishpacha_villa_4
			has_character_modifier = jip_mishpacha_villa_5
		}
	}
	effect = {
		jip_mishpacha_refresh_villa_modifier_effect = yes
	}
}

# root = the domicile, scope:owner = its owner
on_domicile_building_completed = {
	on_actions = {
		jip_mishpacha_domicile_building_completed_on_action
	}
}

jip_mishpacha_domicile_building_completed_on_action = {
	trigger = {
		scope:owner ?= { has_government = mishpacha_government }
	}
	effect = {
		scope:owner = { jip_mishpacha_refresh_villa_modifier_effect = yes }
	}
}

# Mishpacha contracts (common/task_contracts/jip_mishpacha_contracts.txt): player families get a fresh batch of offers each
# year, AI families a hidden roll in their place
# root = the family head
jip_mishpacha_contracts_on_action = {
	trigger = {
		has_government = mishpacha_government
	}
	effect = {
		if = {
			limit = { is_ai = yes }
			jip_mishpacha_ai_contract_effect = yes
		}
		else = {
			jip_mishpacha_refresh_contracts_effect = yes
		}
	}
}

# New offers when the estate settles somewhere new or a new head takes over; the variable keeps an inheritance of several
# titles to one batch
# root = the family head
jip_mishpacha_contracts_new_area_on_action = {
	trigger = {
		jip_mishpacha_contract_owner_trigger = yes
		NOT = { has_variable = jip_mishpacha_contracts_refreshed }
	}
	effect = {
		set_variable = {
			name = jip_mishpacha_contracts_refreshed
			value = yes
			days = 30
		}
		jip_mishpacha_refresh_contracts_effect = yes
	}
}

# Game start: Jewish courtiers for the families, so their court positions and contract schemes have people to fill them
# root = the family head
jip_mishpacha_game_start_courtiers_on_action = {
	trigger = {
		has_government = mishpacha_government
	}
	effect = {
		jip_mishpacha_generate_courtiers_effect = yes
	}
}
