# Character converts faith through the convert window or the matching script effect
# root = the converted character
on_faith_conversion = {
	on_actions = {
		# Delayed so vanilla's faith_conversion.0003 has converted the family first, keeping them out of the heir pick
		delay = { days = 1 }
		mishpacha_left_judaism_on_action
	}
}

# Yearly safety net for conversions that bypass on_faith_conversion, e.g. set_character_faith in events
# root = the character
yearly_playable_pulse = {
	on_actions = {
		mishpacha_left_judaism_on_action
		jip_mishpacha_ai_migration_on_action
		jip_mishpacha_succession_law_on_action
	}
}

mishpacha_left_judaism_on_action = {
	trigger = {
		has_government = mishpacha_government
		NOT = { religion = religion:judaism_religion }
	}

	effect = {
		jip_mishpacha_leave_faith_effect = yes
	}
}

# An expelled family reaches its refuge
# root = the family head
jip_expulsion_refuge_arrival_on_action = {
	trigger = {
		has_variable = jip_expulsion_refuge
	}
	effect = {
		jip_mishpacha_settle_refuge_effect = yes
	}
}

# A fleeing family's journey was cut short (captured, killed, cancelled): their estate is still moved out
# root = the travelling character
on_travel_plan_abort = {
	on_actions = {
		jip_expulsion_refuge_arrival_on_action
	}
}

# AI families now and then weigh moving their estate to a more welcoming realm (about every three years)
# root = the family head
jip_mishpacha_ai_migration_on_action = {
	trigger = {
		jip_mishpacha_ai_can_migrate_trigger = yes
	}
	effect = {
		random = {
			chance = 33
			jip_mishpacha_ai_migration_effect = yes
		}
	}
}

# Safety net: a Mishpacha family head without a realm succession law can't have a succession built
# root = the character
jip_mishpacha_succession_law_on_action = {
	trigger = {
		has_government = mishpacha_government
		NOT = { has_realm_law = mishpacha_succession_law }
	}
	effect = {
		jip_mishpacha_ensure_succession_law_effect = yes
	}
}
