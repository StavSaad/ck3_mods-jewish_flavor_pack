# A title is inherited by a character
# root = the new holder
# scope:title = the title that changes hands
# scope:previous_holder = previous holder. Should be dead

on_title_gain_inheritance = {
	on_actions = {
		exilarchate_inheritance_on_action
	}
}

exilarchate_inheritance_on_action = {
	trigger = {
		OR = {
			scope:title = title:d_exilarchate
			scope:title = title:d_karaite_exilarchate
		}
		scope:title = {
			is_landless_type_title = yes
		}
	}

	effect = {
		scope:title = {
			set_capital_county = root.primary_title.title_capital_county
			holder.domicile ?= {
				move_domicile = root.primary_title.title_capital_county.title_province
			}
		}
	}
}

# A title is transferred to a new character
# root = the new holder
# scope:title = the title that changes hands
on_title_gain = {
	on_actions = {
		# Delayed so a granted title's new liege relation is in place first
		delay = { days = 1 }
		mishpacha_landed_government_switch_on_action
		jip_mishpacha_villa_on_action # Villa prestige for a family head who inherits the estate
		jip_mishpacha_contracts_new_area_on_action # Offers for a new family head
		jip_carthage_restored_on_action
	}
}

# Canaanism can be converted to once Carthage is restored: only vanilla's Restore Carthage hands out k_carthage
jip_carthage_restored_on_action = {
	trigger = {
		scope:title = title:k_carthage
	}
	effect = {
		jip_unlock_canaanite_conversion_effect = yes
	}
}

mishpacha_landed_government_switch_on_action = {
	trigger = {
		has_government = familial_government
		is_landed = yes
	}

	effect = {
		jip_mishpacha_adopt_landed_government_effect = yes
	}
}

# A character is about to die, still holding their titles
# root = the dying character
on_death = {
	on_actions = {
		jip_kohen_gadol_death_on_action
	}
}

# Marks a High Priest who dies in office, so their portrait keeps the Kohen Gadol garments (jip_former_kohen_gadol_trigger)
jip_kohen_gadol_death_on_action = {
	trigger = {
		has_title = title:d_kohen_gadol
	}
	effect = {
		# A global list, since dead characters can't hold variables
		add_to_global_variable_list = {
			name = jip_former_kohen_gadols
			target = this
		}
	}
}
