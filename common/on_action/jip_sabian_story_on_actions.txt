# The Return of the Moon God, the Sabian story (common/story_cycles/jip_story_sabian_revival.txt): it starts before the
# lobby for the head of Thabit ibn Qurra's house (d_jip_harrani, held from 867), so the lobby already shows his secret
# faith; a player head is told the story as the game begins (jip_sabian_story.1001)
on_game_start = {
	on_actions = {
		jip_sabian_story_game_start_on_action
	}
}

on_game_start_after_lobby = {
	on_actions = {
		jip_sabian_story_game_start_after_lobby_on_action
	}
}

jip_sabian_story_game_start_on_action = {
	effect = {
		jip_sabian_story_start_effect = yes
	}
}

jip_sabian_story_game_start_after_lobby_on_action = {
	effect = {
		# A player of the old faiths (a ruler-designer character) finds them open to conversion from day 0. A Mesopotamian
		# pagan player already has what the Sabian story works for, so the story ends, and is never started or offered
		if = {
			limit = {
				any_player = { faith = faith:mesopotamian_pagan }
			}
			jip_unlock_mesopotamian_conversion_effect = yes
			set_global_variable = jip_sabian_story_disabled
			title:d_jip_harrani.holder ?= {
				random_owned_story = {
					type = story_jip_sabian_revival
					end_story = yes
				}
			}
		}
		if = {
			limit = {
				any_player = { faith = faith:semitic_pagan }
			}
			jip_unlock_canaanite_conversion_effect = yes
		}
		# In case the head changed in the lobby
		jip_sabian_story_start_effect = yes
		title:d_jip_harrani.holder ?= {
			# The patron, now that the start-date lieges are set
			random_owned_story = {
				type = story_jip_sabian_revival
				jip_sabian_story_update_patron_effect = yes
			}
			if = {
				limit = {
					is_ai = no
					jip_owns_sabian_story_trigger = yes
				}
				trigger_event = jip_sabian_story.1001
			}
		}
		# Any other Sabian player (a ruler-designer character, or the head of a Sabian family) may take up the story
		every_player = {
			limit = {
				jip_sabian_story_can_be_offered_trigger = yes
			}
			trigger_event = jip_sabian_story.1002
		}
	}
}

# The story owner who holds Harran in Act I or II, however they came by it, moves on to Act III, or to Act IV if
# independent (jip_sabian_story_harran_held_effect; the story's monthly check catches the rest)
on_title_gain = {
	on_actions = {
		jip_sabian_story_harran_gained_on_action
	}
}

# root = the new holder, scope:title = the title
jip_sabian_story_harran_gained_on_action = {
	trigger = {
		scope:title = title:c_harran
		OR = {
			jip_sabian_story_in_act_trigger = { ACT = 1 }
			jip_sabian_story_in_act_trigger = { ACT = 2 }
		}
	}
	effect = {
		jip_sabian_story_harran_held_effect = yes
	}
}

# A Sabian scholar offers a Muslim ruler their learning (jip_sabian.0200; 0.15 plan, section 2), about every four years
yearly_playable_pulse = {
	on_actions = {
		jip_sabian_scholar_on_action
	}
}

jip_sabian_scholar_on_action = {
	trigger = {
		jip_can_patronise_sabian_scholar_trigger = yes
	}
	effect = {
		random = {
			chance = 25
			trigger_event = {
				id = jip_sabian.0200
				days = { 30 300 }
			}
		}
	}
}
