# Before the lobby, so they can be played; the after-lobby setup gives them courtiers and contracts like any family.

	trigger = {
		# Estates are Roads to Power
		has_ep3_dlc_trigger = yes
	}
	effect = {
		jip_spawn_jewish_families_effect = yes
	}
}
