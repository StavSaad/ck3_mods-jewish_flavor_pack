on_game_start = {
    on_actions = {
		jip_867_effect_setup
		jip_1066_effect_setup
        jip_1178_effect_setup
		jip_liege_effect_setup
		jip_jewish_families_game_start_on_action
		jip_rish_amma_game_start_on_action
		jip_masada_game_start_on_action
    }
}

# The ruins of Masada at every start date, in the special district slot province history gives Zughar
jip_masada_game_start_on_action = {
	effect = {
		title:b_zughar.title_province ?= {
			if = {
				limit = { NOT = { has_building_or_higher = masada_01 } }
				add_special_district = masada_01_district
			}
		}
	}
}

# The Mandaeans' head of faith, at every start date (jip_rish_amma_setup_effect)
jip_rish_amma_game_start_on_action = {
	effect = {
		jip_rish_amma_setup_effect = yes
	}
}

# New Mishpacha families across Europe, the Middle East and North Africa (jip_spawn_jewish_families_effect), then a few
# Sabian and Mandaean families (jip_spawn_gnostic_families_effect).
# Before the lobby, so they can be played; the after-lobby setup gives them courtiers and contracts like any family.
jip_jewish_families_game_start_on_action = {
	trigger = {
		jip_familial_estates_available_trigger = yes
	}
	effect = {
		jip_spawn_jewish_families_effect = yes
		jip_spawn_gnostic_families_effect = yes
	}
}

jip_liege_effect_setup = {
	effect = {
		jip_liege_setup_effect = yes
		every_ruler = {
			limit = { has_government = familial_government }
			jip_mishpacha_ensure_succession_law_effect = yes
			jip_familial_record_religion_effect = yes
			# Family titles are house titles (title_type_noble_family): their holders lead their houses, as vanilla's
			# noble_family_title_realm_setup_effect makes sure for its own historical families
			save_temporary_scope_as = jip_family_holder
			if = {
				limit = {
					any_held_title = { jip_is_mishpacha_family_title_trigger = yes }
					exists = house
					NOT = { house.house_head ?= this }
				}
				house = { set_house_head = scope:jip_family_holder }
			}
			# Every family takes the lord of its estate's city as patron (0.16), whoever the setup above named
			if = {
				limit = { is_landed = no }
				jip_mishpacha_join_estate_realm_if_new_effect = yes
			}
		}
	}
}
