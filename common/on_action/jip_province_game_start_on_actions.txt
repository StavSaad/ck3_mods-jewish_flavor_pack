on_game_start = {
    on_actions = {
		jip_867_effect_setup
		jip_1066_effect_setup
        jip_1178_effect_setup
		jip_liege_effect_setup
		jip_jewish_families_game_start_on_action
    }
}

# New Mishpacha families across Europe, the Middle East and North Africa (jip_spawn_jewish_families_effect).
# Before the lobby, so they can be played; the after-lobby setup gives them courtiers and contracts like any family.
jip_jewish_families_game_start_on_action = {
	trigger = {
		# Estates are Roads to Power
		has_ep3_dlc_trigger = yes
	}
	effect = {
		jip_spawn_jewish_families_effect = yes
	}
}

jip_liege_effect_setup = {
	effect = {
		jip_liege_setup_effect = yes
		every_ruler = {
			limit = { has_government = mishpacha_government }
			jip_mishpacha_ensure_succession_law_effect = yes
			# A family whose estate lies under a duke or lesser ruler leaves any other liege and pays them tribute
			save_scope_as = jip_family_head
			if = {
				limit = {
					is_landed = no
					domicile.domicile_location.county.holder.top_liege ?= {
						this != scope:jip_family_head
						highest_held_title_tier <= tier_duchy
						NOT = { scope:jip_family_head.suzerain ?= this }
					}
				}
				domicile.domicile_location.county.holder.top_liege = { save_scope_as = jip_estate_ruler }
				jip_mishpacha_change_liege_effect = { LIEGE = scope:jip_estate_ruler }
			}
		}
	}
}
