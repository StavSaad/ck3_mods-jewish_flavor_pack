	effect = {
		jip_akitu_neglect_effect = yes
	}
}

# Every Jewish playable character keeps a household seder each Pesach (jip_feast.5020), sent to land in April; it is
# skipped by whoever sat at a grand seder that spring (jip_kept_pesach)
jip_household_seder_on_action = {
	trigger = {
		religion = religion:judaism_religion
	}
	effect = {
		trigger_event = {
			id = jip_feast.5020
			days = { jip_days_to_pesach_value jip_days_to_pesach_late_value }
		}
	}
}
