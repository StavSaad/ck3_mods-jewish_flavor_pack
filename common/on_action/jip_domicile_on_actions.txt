on_domicile_moved = {
	on_actions = {
		jewish_admin_migration_on_action
	}
}

jewish_admin_migration_on_action = {
	trigger = {
		exists = scope:owner
		scope:owner = {
			has_government = mishpacha_government
		}
	}

	effect = {
		domicile_location.county.holder.top_liege = {
			save_scope_as = recipient_liege
		}
		scope:owner = {
			break_subject_contract_and_establish_tributary_effect = { SUZERAIN = scope:recipient_liege TRIBUTARY = this }
		}
	}
}
