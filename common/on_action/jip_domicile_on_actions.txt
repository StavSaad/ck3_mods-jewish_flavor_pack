# An estate never travels (vanilla's estate domicile type has no travel = yes), so a migrating AI family's head sets
# out alone with the destination in var:jip_mishpacha_migrating_to (jip_mishpacha_ai_migration_effect), and the estate
# is moved there when they arrive. A journey cut short leaves the family where it was.
# root = the travelling character
on_travel_plan_complete = {
	on_actions = {
		jip_familial_migration_arrival_on_action
	}
}

on_travel_plan_abort = {
	on_actions = {
		jip_familial_migration_abort_on_action
	}
}

jip_familial_migration_arrival_on_action = {
	trigger = {
		has_variable = jip_mishpacha_migrating_to
		government_has_flag = government_is_familial
		is_landed = no
		exists = domicile
	}
	effect = {
		jip_mishpacha_settle_estate_on_arrival_effect = yes
	}
}

jip_familial_migration_abort_on_action = {
	trigger = {
		has_variable = jip_mishpacha_migrating_to
	}
	effect = {
		remove_variable = jip_mishpacha_migrating_to
	}
}

# Every move of an estate, by the Move Estate button or by script, sets it down at once and fires on_domicile_moved
# there. The family joins its new realm a day later, so a script that moves the estate and then sets the liege itself
# (game start, expulsion refuge, conversion) goes first.
# root = the domicile, scope:owner = its owner
on_domicile_moved = {
	on_actions = {
		jip_familial_estate_moved_on_action
	}
}

jip_familial_estate_moved_on_action = {
	trigger = {
		exists = scope:owner
		scope:owner = {
			government_has_flag = government_is_familial
			is_landed = no
		}
	}
	effect = {
		scope:owner = {
			trigger_event = {
				on_action = jip_familial_estate_settled_on_action
				days = 1
			}
		}
	}
}

# root = the family head, a day after the estate moved
jip_familial_estate_settled_on_action = {
	trigger = {
		government_has_flag = government_is_familial
		is_landed = no
		exists = domicile
	}
	effect = {
		# Only switch when the estate lies in a different realm: the family leaves its old liege or suzerain and
		# becomes the vassal, or under a duke or lesser ruler the tributary, of the new one
		jip_mishpacha_join_estate_realm_if_new_effect = yes
		# Offers from the new neighbourhood
		trigger_event = { on_action = jip_mishpacha_contracts_new_area_on_action }
	}
}
