on_domicile_moved = {
	effect = {
		if = {
			limit = {
				exists = scope:owner
				scope:owner = {
					has_government = jewish_admin_family_government
				}
				domicile_location.county.holder.top_liege = {
					primary_title.tier >= tier_kingdom
				}
			}
			domicile_location.county.holder.top_liege = {
				save_scope_as = recipient_liege
			}
			create_title_and_vassal_change = {
				type = swear_fealty
				save_scope_as = change
				add_claim_on_loss = no
			}
			change_liege = {
				liege = scope:recipient_liege
				change = scope:change
			}
			scope:owner = {
				resolve_title_and_vassal_change = scope:change
			}
		}
		if = {
			limit = {
				exists = scope:owner
			}
			scope:owner = { remove_variable = ep3_laamp_decision_1000_is_visiting }
		}
	}
}


# root = domicile
# scope:owner = <character> owner of domicile
on_domicile_building_started = {
	effect = {
		scope:owner = {
			send_interface_message = {
				type = msg_domicile_building_started
				left_icon = this
			}
		}
		if = {
			limit = {
				scope:owner = { has_character_modifier = tgp_subsidised_estate_buildings_modifier }
				has_variable = subsidised_estate_var 
				var:subsidised_estate_var > 1
			}
			change_variable = {
				name = subsidised_estate_var
				subtract = 1
			}
			scope:owner = {
				remove_character_modifier = tgp_subsidised_estate_buildings_modifier
				add_character_modifier = tgp_subsidised_estate_buildings_modifier
			}
		}
		else = {
			scope:owner = {
				remove_character_modifier = tgp_subsidised_estate_buildings_modifier
			}
			remove_variable = subsidised_estate_var
		}
	}
}

# root = domicile
# scope:owner = <character> owner of domicile
on_domicile_building_completed = {
	effect = {
		scope:owner ?= {
			send_interface_message = {
				type = msg_domicile_building_completed
				left_icon = this
			}
			#Cultivated Sophistication cultural tradition bonus
			if = {
				limit = {
					culture = {
						has_cultural_parameter = estate_building_upgrade_development_gain
					}
					is_landed = yes
					scope:owner.capital_county ?= root.domicile_location.county
				}
				root = {
					domicile_location.county = {
						change_development_progress = cultivated_sophistication_development_gain
					}
				}
			}
			# Building Oath
			if = {
				limit = {
					has_variable = stewardship_oath_2_buildings
				}
				change_variable = {
					name = stewardship_oath_2_buildings
					add = 1
				}
			}
		}
	}
}
