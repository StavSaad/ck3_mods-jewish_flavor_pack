	is_shown = {
		dynasty = {
			OR = {

# The Harranian legacy (0.15 plan, section 10): for dynasties whose dynast is a Sabian or of the old faith, marked
# var:jip_harranian_dynasty by jip_update_harranian_dynasty_effect (common/on_action/jip_legacy_on_actions.txt). Never name a
# faith or rite here or in the perks: legacies load before the faiths, and the game crashes at startup
harranian_legacy_track = {
	is_shown = {
		dynasty = {
			OR = {
				has_variable = jip_harranian_dynasty
				has_dynasty_perk = harranian_legacy_1
			}
		}
	}
}
