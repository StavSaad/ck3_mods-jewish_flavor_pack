# Game start: up to 40 new Mishpacha families across Europe, the Middle East and North Africa, so the Jewish world has
# more houses to trade, marry and compete with. Each region gets a share; a region with fewer welcoming cities than its
# share (jip_jewish_family_home_county_trigger) gets fewer families. One family per county.
# Courtiers, contracts and estate fixes come from the after-lobby setup like any other family.
# Runs in on_game_start, where root is empty: nothing here may use root.
jip_spawn_jewish_families_effect = {
	# Counties that already hold a family estate are taken
	every_ruler = {
		limit = { has_government = familial_government }
		domicile ?= {
			domicile_location.county = { add_to_list = jip_family_counties }
		}
	}
	# So are the counties of the game's Jewish merchant companies (1.21: about 75 at 867, most of them Radhanite, from
	# France to Samarkand), which also count against their region's share (jip_spawn_jewish_families_in_region_effect)
	every_ruler = {
		limit = { jip_is_jewish_merchant_trigger = yes }
		if = {
			limit = { exists = domicile }
			domicile.domicile_location.county = {
				add_to_list = jip_family_counties
				add_to_list = jip_jewish_company_counties
			}
		}
		else_if = {
			limit = { exists = location }
			location.county = {
				add_to_list = jip_family_counties
				add_to_list = jip_jewish_company_counties
			}
		}
	}
	# Up to 40 in all (60 before 1.21), less half a family for each Jewish company in the region
	jip_spawn_jewish_families_in_region_effect = { REGION = world_europe_west_iberia COUNT = 7 }
	jip_spawn_jewish_families_in_region_effect = { REGION = world_europe_west_francia COUNT = 4 }
	jip_spawn_jewish_families_in_region_effect = { REGION = world_europe_west_germania COUNT = 4 }
	jip_spawn_jewish_families_in_region_effect = { REGION = world_europe_south_italy COUNT = 5 }
	jip_spawn_jewish_families_in_region_effect = { REGION = world_europe_east COUNT = 1 }
	jip_spawn_jewish_families_in_region_effect = { REGION = world_europe_south_east COUNT = 2 }
	jip_spawn_jewish_families_in_region_effect = { REGION = world_asia_minor COUNT = 2 }
	jip_spawn_jewish_families_in_region_effect = { REGION = world_steppe_west COUNT = 1 }
	jip_spawn_jewish_families_in_region_effect = { REGION = world_middle_east_jerusalem COUNT = 3 }
	jip_spawn_jewish_families_in_region_effect = { REGION = world_middle_east_arabia COUNT = 3 }
	jip_spawn_jewish_families_in_region_effect = { REGION = world_middle_east_persia COUNT = 2 }
	jip_spawn_jewish_families_in_region_effect = { REGION = world_africa_north_west COUNT = 3 }
	jip_spawn_jewish_families_in_region_effect = { REGION = world_africa_north_east COUNT = 3 }
}

# Up to $COUNT$ new families in $REGION$, each in a different county, weighted by jip_jewish_family_county_weight_value:
# half a family fewer for each Jewish merchant company in the region, but at least one. The share is a global variable,
# since root is empty here
jip_spawn_jewish_families_in_region_effect = {
	every_county_in_region = {
		region = $REGION$
		limit = { jip_jewish_family_home_county_trigger = yes }
		add_to_list = jip_family_candidates_$REGION$
	}
	set_global_variable = { name = jip_region_families_left value = $COUNT$ }
	every_in_list = {
		list = jip_jewish_company_counties
		limit = {
			title_province = { geographical_region = $REGION$ }
		}
		change_global_variable = { name = jip_region_families_left add = -0.5 }
	}
	if = {
		limit = { global_var:jip_region_families_left < 1 }
		set_global_variable = { name = jip_region_families_left value = 1 }
	}
	while = {
		count = $COUNT$
		if = {
			limit = {
				global_var:jip_region_families_left >= 1
				any_in_list = {
					list = jip_family_candidates_$REGION$
					NOT = { is_in_list = jip_family_counties }
				}
			}
			random_in_list = {
				list = jip_family_candidates_$REGION$
				limit = {
					NOT = { is_in_list = jip_family_counties }
				}
				weight = {
					base = 1
					modifier = {
						add = jip_jewish_family_county_weight_value
					}
				}
				add_to_list = jip_family_counties
				random_county_province = {
					limit = { has_holding_type = city_holding }
					save_scope_as = jip_new_family_home
				}
			}
			jip_create_jewish_family_effect = yes
			change_global_variable = { name = jip_region_families_left add = -1 }
		}
	}
	remove_global_variable = jip_region_families_left
}

# A new Mishpacha family in the city scope:jip_new_family_home (jip_create_familial_family_effect)
jip_create_jewish_family_effect = {
	scope:jip_new_family_home = {
		jip_jewish_family_culture_effect = yes
		jip_jewish_family_faith_effect = yes
	}
	jip_create_familial_family_effect = yes
}

# A new Familial family of any faith in the city scope:jip_new_family_home, of the culture, faith and rite saved as
# scope:jip_new_family_culture, scope:jip_new_family_faith and scope:jip_new_family_rite: a head aged 20 to 40, a family
# title and estate, the local ruler as liege or suzerain, a spouse and children
jip_create_familial_family_effect = {
	scope:jip_new_family_home = {
		county.holder.top_liege = { save_scope_as = jip_new_family_liege }
	}
	create_character = {
		# Nothing Jewish in it but the name: skills and traits of a family of merchants and scholars
		template = jip_jewish_family_head_template
		location = scope:jip_new_family_home
		culture = scope:jip_new_family_culture
		faith = scope:jip_new_family_faith
		rite = scope:jip_new_family_rite
		# Also the name jip_mishpacha_house_title_name reads
		save_scope_as = jip_new_mishpacha_head
	}
	scope:jip_new_mishpacha_head = {
		# Same steps as a dispossessed lord (jip_jewish_lord_lose_lands_effect): the estate comes with a domicile title,
		# then the noble family title takes its place
		create_domicile_title = {
			name = jip_mishpacha_house_title_name
			holder = scope:jip_new_mishpacha_head
			government = familial_government
			tier = duchy
			save_scope_as = jip_domicile_title
		}
		give_noble_family_title = {
			name = noble_family_name
			tier = duchy
			article = DEFAULT_TITLE_NAME_ARTICLE
			government = familial_government
			save_scope_as = jip_new_family_title
		}
		destroy_title = scope:jip_domicile_title
		domicile ?= {
			move_domicile = scope:jip_new_family_home
			jip_add_random_family_estate_building_effect = yes
		}
		jip_mishpacha_change_liege_effect = { LIEGE = scope:jip_new_family_liege }
		if = {
			limit = { NOT = { has_government = familial_government } }
			change_government = familial_government
		}
		jip_mishpacha_ensure_succession_law_effect = yes
		add_gold = jip_jewish_family_starting_gold_value
		# Vanilla's generated families: a spouse of the same age and up to three children, by the mother's age
		game_rule_create_spouse_and_children = { CHARACTER = this LOCATION = this }
	}
}

# Province scope: the Jewish culture of the community a new family joins, saved as scope:jip_new_family_culture.
# Iberia is Sephardi, Latin Europe Ashkenazi, the Khazar steppe Radhanite, the Levant, Egypt and the Maghreb
# Musta'arabi, and the rest of the Caliphate and Byzantium Bavlim. Eastern Europe and Persia, on the Radhanite routes, mix in Radhanites.
jip_jewish_family_culture_effect = {
	if = {
		limit = { geographical_region = world_europe_west_iberia }
		culture:sephardi = { save_scope_as = jip_new_family_culture }
	}
	else_if = {
		limit = {
			OR = {
				geographical_region = world_europe_west
				geographical_region = world_europe_south_italy
			}
		}
		culture:ashkenazi = { save_scope_as = jip_new_family_culture }
	}
	else_if = {
		limit = { geographical_region = world_steppe_west }
		culture:radhanite = { save_scope_as = jip_new_family_culture }
	}
	else_if = {
		limit = { geographical_region = world_europe_east }
		random_list = {
			60 = { culture:ashkenazi = { save_scope_as = jip_new_family_culture } }
			40 = { culture:radhanite = { save_scope_as = jip_new_family_culture } }
		}
	}
	# The Arabic-speaking Jews of the Levant, Egypt and the Maghreb: vanilla's Musta'arabi (1.21), as its own Egyptian,
	# Maghrebi and Yemeni Jews
	else_if = {
		limit = {
			OR = {
				geographical_region = world_middle_east_jerusalem
				geographical_region = world_africa_north_east
				geographical_region = world_africa_north_west
			}
		}
		culture:mustaarabi = { save_scope_as = jip_new_family_culture }
	}
	else_if = {
		limit = { geographical_region = world_middle_east_persia }
		random_list = {
			70 = { culture:bavlim = { save_scope_as = jip_new_family_culture } }
			30 = { culture:radhanite = { save_scope_as = jip_new_family_culture } }
		}
	}
	else = {
		culture:bavlim = { save_scope_as = jip_new_family_culture }
	}
}

# Province scope: the faith and rite of a new family, saved as scope:jip_new_family_faith and scope:jip_new_family_rite.
# Mostly Rabbinic, a few of them Merkabah mystics; Karaites where their communities were strong: Egypt, the Caliphate,
# Byzantium and the Khazar lands
jip_jewish_family_faith_effect = {
	random_list = {
		85 = {
			faith:rabbinic_faith = { save_scope_as = jip_new_family_faith }
			rite:rabbinism = { save_scope_as = jip_new_family_rite }
		}
		5 = {
			faith:rabbinic_faith = { save_scope_as = jip_new_family_faith }
			rite:merkabah = { save_scope_as = jip_new_family_rite }
		}
		0 = {
			modifier = {
				add = 15
				OR = {
					geographical_region = world_middle_east
					geographical_region = world_europe_south_east
					geographical_region = world_asia_minor
				}
			}
			modifier = {
				add = 25
				geographical_region = world_africa_north_east
			}
			modifier = {
				add = 35
				geographical_region = world_steppe_west
			}
			faith:karaism = { save_scope_as = jip_new_family_faith }
			rite:karaism = { save_scope_as = jip_new_family_rite }
		}
	}
}

# Domicile scope: one starting room for a new family's estate, as vanilla's add_random_internal_estate_building does for
# landless nobles, but with the families' own lines (the Clerk's Office and Hachnasat Orchim copies) and without the
# Wine Cellar, which families don't build, or the Prison, whose vanilla option logs an error on a new estate (1.21)
jip_add_random_family_estate_building_effect = {
	random_list = {
		10 = {

			trigger = {
