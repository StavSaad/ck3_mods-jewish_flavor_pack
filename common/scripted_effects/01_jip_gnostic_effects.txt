# Mandaeans and Sabians (mandaean_faith)

# Gives the Rish Amma (d_rish_amma), head of faith of the Mandaeans, a holder at game start, at any start date, unless
# history or the engine already has one. On death the engine picks the next head, as for other heads of faith.
# Runs in on_game_start: root is empty
jip_rish_amma_setup_effect = {
	if = {
		limit = {
			NOT = { exists = title:d_rish_amma.holder }
		}
		create_character = {
			template = jip_rish_amma_template
			location = title:c_tib.title_province
			culture = culture:nabati
			faith = faith:mandaean_faith
			rite = rite:mandeaism
			save_scope_as = jip_new_rish_amma
		}
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:d_rish_amma = {
			change_title_holder = {
				holder = scope:jip_new_rish_amma
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
}

# Character scope: the Ginza Rabba copied (jip_mandaean_copy_ginza_decision) by the most learned of the scribes at hand: the
# character, their Ganzibra or their Mandaean court chaplain. The copyist signs the colophon (the artifact's creator)
jip_mandaean_copy_ginza_effect = {
	if = {
		limit = { jip_mandaean_ginza_scribe_trigger = yes }
		add_to_temporary_list = jip_ginza_scribes
	}
	every_court_position_holder = {
		type = jip_ganzibra_court_position
		limit = { jip_mandaean_ginza_scribe_trigger = yes }
		add_to_temporary_list = jip_ginza_scribes
	}
	cp:councillor_court_chaplain ?= {
		if = {
			limit = { jip_mandaean_ginza_scribe_trigger = yes }
			add_to_temporary_list = jip_ginza_scribes
		}
	}
	ordered_in_list = {
		list = jip_ginza_scribes
		order_by = learning
		save_scope_as = jip_ginza_copyist
	}
	create_artifact = {
		name = jip_artifact_ginza_rabba_name
		description = jip_artifact_ginza_rabba_desc
		creator = scope:jip_ginza_copyist
		visuals = book
		type = chronicle # a book slot artifact with no required features (vanilla's book type needs a subject and cover)
		template = jip_ginza_rabba_template
		modifier = jip_ginza_rabba_modifier
		wealth = 40
		quality = 60
		decaying = yes
		save_scope_as = jip_ginza
	}
	scope:jip_ginza = { set_variable = jip_ginza_rabba }
	if = {
		limit = { NOT = { scope:jip_ginza_copyist = root } }
		scope:jip_ginza_copyist = {
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = 15
			}
		}
	}
}

# The Rish Amma's title passed (jip_rish_amma_succession_on_action, root = the new holder): if a Ganzibra (Mandaean family
# court position) is alive, the most learned of them takes it instead, as the rish amma rose from the ganzibras
jip_rish_amma_succession_effect = {
	every_ruler = {
		limit = { employs_court_position = jip_ganzibra_court_position }
		every_court_position_holder = {
			type = jip_ganzibra_court_position
			limit = {
				is_adult = yes
				is_imprisoned = no
				rite = rite:mandeaism
			}
			add_to_temporary_list = jip_ganzibras
		}
	}
	if = {
		limit = {
			any_in_list = {
				list = jip_ganzibras
				count >= 1
			}
		}
		ordered_in_list = {
			list = jip_ganzibras
			order_by = learning
			save_scope_as = jip_ganzibra
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = change
			add_claim_on_loss = no
		}
		title:d_rish_amma = {
			change_title_holder = {
				holder = scope:jip_ganzibra
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
}

# Dynasty scope: marks a dynasty whose dynast is a Sabian or of the old faith (var:jip_harranian_dynasty), the mark the
# Harranian legacy checks (common/on_action/jip_legacy_on_actions.txt). Legacies and perks load before the faiths, so their
# script can't name a faith or rite: rite:sabianism there crashed the game at startup
jip_update_harranian_dynasty_effect = {
	if = {
		limit = {
			dynast ?= { jip_is_sabian_rite_trigger = yes }
		}
		set_variable = jip_harranian_dynasty
	}
	else_if = {
		limit = { has_variable = jip_harranian_dynasty }
		remove_variable = jip_harranian_dynasty
	}
}
