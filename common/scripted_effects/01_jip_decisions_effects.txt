handle_maa_traditions_for_culture = {
	if = {
		limit = {
			scope:$CULTURE$_former.culture = {
				has_cultural_tradition = tradition_$TRADITION$
			}
		}
		culture:$CULTURE$ = {
			add_culture_tradition = tradition_$TRADITION$
		}
	}
}

### Land by region (jip_geographical_regions.txt): a map mod's compatch changes the regions, not these

# Character scope (the founder of the kingdom $KINGDOM$): the land of $REGION$ joins it. A duchy joins whole by the
# rule $DUCHIES$: capital (its de jure capital lies in the region; Phoenicia, so Antioch comes with Alexandretta) or
# whole (all its de jure counties lie in the region; Edom, so the Negev stays with Oultrejourdain). The region's other
# counties join the duchy $HOME$ (one that joins)
jip_kingdom_takes_region_effect = {
	every_county_in_region = {
		region = $REGION$
		limit = {
			duchy ?= {
				title_capital_county ?= prev
				jip_duchy_joins_region_by_$DUCHIES$_trigger = { REGION = $REGION$ }
			}
		}
		duchy = { set_de_jure_liege_title = title:$KINGDOM$ }
	}
	every_county_in_region = {
		region = $REGION$
		limit = {
			NOT = { de_jure_liege ?= title:$HOME$ }
			NOT = {
				duchy ?= { jip_duchy_joins_region_by_$DUCHIES$_trigger = { REGION = $REGION$ } }
			}
		}
		set_de_jure_liege_title = title:$HOME$
	}
}

# Character scope (the founder of the empire $EMPIRE$): every kingdom they hold whose de jure capital lies in $REGION$
# becomes its de jure kingdom, whatever empire it belonged to
jip_empire_takes_held_kingdoms_effect = {
	every_held_title = {
		limit = {
			tier = tier_kingdom
			jip_title_capital_in_region_trigger = { REGION = $REGION$ }
		}
		set_de_jure_liege_title = title:$EMPIRE$
	}
}

# Character scope (the founder of the empire $EMPIRE$): as above, for the kingdoms they also completely control
jip_empire_takes_held_controlled_kingdoms_effect = {
	save_scope_as = jip_empire_founder
	every_held_title = {
		limit = {
			tier = tier_kingdom
			jip_title_capital_in_region_trigger = { REGION = $REGION$ }
			save_temporary_scope_as = jip_empire_kingdom
			scope:jip_empire_founder = { completely_controls = scope:jip_empire_kingdom }
		}
		set_de_jure_liege_title = title:$EMPIRE$
	}
}

# Character scope (the founder of the empire $EMPIRE$): every kingdom that exists, that they completely control and
# whose de jure capital lies in $REGION$ becomes its de jure kingdom, whoever holds it. Found through its capital
# county, which lies in the founder's realm
jip_empire_takes_controlled_kingdoms_effect = {
	save_scope_as = jip_empire_founder
	every_sub_realm_county = {
		limit = {
			kingdom ?= {
				title_capital_county ?= prev
				is_title_created = yes
				jip_title_capital_in_region_trigger = { REGION = $REGION$ }
				save_temporary_scope_as = jip_empire_kingdom
				scope:jip_empire_founder = { completely_controls = scope:jip_empire_kingdom }
			}
		}
		kingdom = { set_de_jure_liege_title = title:$EMPIRE$ }
	}
}

# Character scope (the decision taker): the county they hold where the city $CITY$ lies (jip_city_<name>, a one-province
# region) goes to a new republic, its city the county's capital
reestablish_canaanite_city_effect = {
	random_held_title = {
		limit = {
			tier = tier_county
			jip_county_has_city_trigger = { CITY = $CITY$ }
		}
		save_scope_as = jip_canaanite_city_county
		random_county_province = {
			limit = { geographical_region = $CITY$ }
			barony = { save_scope_as = jip_canaanite_city_barony }
		}
	}
	custom_description = {
		text = reestablish_canaanite_city_decision_effect
		object = scope:jip_canaanite_city_barony
		create_character = {
			random_traits = yes
			culture = root.culture
			faith = root.faith
			gender = male
			location = root.location
			save_scope_as = new_owner
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:jip_canaanite_city_county = {
			change_title_holder = {
				holder = scope:new_owner
				change = scope:change
				take_baronies = yes
			}
		}
		resolve_title_and_vassal_change = scope:change
		scope:new_owner = {
			change_government = republic_government
		}
	}
	scope:jip_canaanite_city_barony = { set_capital_barony = yes }
}

# The gods of Aram restored (jip_restore_gods_of_aram_decision, jip_palmyra.0002): the ruler and their capital take
# $RITE$ (hadadism, or belism at Palmyra), and Palmyra too if they hold it; the faith opens, and the Christian and
# Muslim rulers around resent it
jip_restore_gods_of_aram_effect = {
	set_character_faith = faith:aramean_pagan
	set_character_rite = rite:$RITE$
	capital_county ?= {
		set_county_faith = faith:aramean_pagan
		set_county_rite = rite:$RITE$
	}
	every_county_in_region = {
		region = jip_city_palmyra
		limit = {
			holder ?= { target_is_same_character_or_above = root }
		}
		set_county_faith = faith:aramean_pagan
		set_county_rite = rite:belism
	}
	jip_unlock_aramean_conversion_effect = yes
	add_piety = major_piety_gain
	save_scope_as = jip_aram_restorer
	custom_description = {
		text = jip_neighbours_resent_gods_of_aram
		object = scope:jip_aram_restorer
		every_neighboring_and_across_water_top_liege_realm_owner = {
			limit = {
				OR = {
					religion = religion:islam_religion
					religion = religion:christianity_religion
				}
			}
			add_opinion = {
				target = scope:jip_aram_restorer
				modifier = jip_restored_old_gods_opinion
			}
			faith = { add_to_temporary_list = jip_angered_faiths }
		}
		every_in_list = {
			list = jip_angered_faiths
			change_fervor = {
				value = slight_fervor_gain
				desc = fervor_gain_jip_gods_of_aram_restored
			}
		}
	}
}

		rite:ashurism = { save_scope_as = jip_old_gods_rite }
	}
	else = {
		rite:sinism = { save_scope_as = jip_old_gods_rite }
	}
	set_character_rite = scope:jip_old_gods_rite
	capital_county ?= {
		set_county_rite = scope:jip_old_gods_rite
	}
	title:c_harran = {
		if = {
			limit = {
				holder ?= { target_is_same_character_or_above = root }
			}
			set_county_rite = scope:jip_old_gods_rite
		}
	}
}
