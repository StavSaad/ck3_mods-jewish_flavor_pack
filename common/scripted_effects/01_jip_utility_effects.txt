	copy_dynasty_perk_effect = { PERK_NAME = $LEGACY_NAME$_legacy_4 }
	copy_dynasty_perk_effect = { PERK_NAME = $LEGACY_NAME$_legacy_5 }
}

copy_dynasty_perk_effect = {
	if = {
		limit = {
			scope:origin.dynasty ?= { has_dynasty_perk = $PERK_NAME$ }
		}
		scope:destination.dynasty ?= {
			add_dynasty_perk = $PERK_NAME$
		}
	}
}

switch_house_effect = {
	custom_description_no_bullet = {
		text = switch_house_effect_desc
		if = {
			limit = {
				house = scope:original_house
			}
			set_house = scope:destination_house
			every_child = {
				switch_house_effect = yes
			}
		}
	}
}
