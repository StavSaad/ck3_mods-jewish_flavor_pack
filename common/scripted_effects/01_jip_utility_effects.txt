# A landed Mishpacha family takes on its liege's government, or feudal when independent
# or when the liege's government doesn't suit a landed vassal.
jip_mishpacha_adopt_landed_government_effect = {
	if = {
		limit = {
			is_independent_ruler = no
			liege ?= {
				NOR = {
					government_has_flag = government_is_mishpacha
					government_has_flag = government_is_theocracy
					government_has_flag = government_is_ecclesiastical
					government_has_flag = government_is_holy_order
					government_has_flag = government_is_mercenary
					government_has_flag = government_is_landless_adventurer
				}
			}
		}
		liege.government_type = { save_scope_as = jip_new_government }
		change_government = scope:jip_new_government
	}
	else = {
		change_government = feudal_government
	}
}


# A Mishpacha family head who leaves Judaism steps down. Their titles pass to the nearest Jewish man of their house:
# children first, then close family, extended family and the rest of the house, eldest first within each group.
# With no such heir, the family title is destroyed and the convert becomes Administrative, keeping any other titles and their liege.
jip_mishpacha_leave_faith_effect = {
	save_scope_as = jip_convert
	if = {
		limit = { is_independent_ruler = no }
		liege = { save_scope_as = jip_convert_liege }
	}
	domicile ?= { domicile_location = { save_scope_as = jip_convert_location } }
	if = {
		limit = {
			house ?= {
				any_house_member = {
					this != scope:jip_convert
					is_male = yes
					religion = religion:judaism_religion
				}
			}
		}
		house = {
			ordered_house_member = {
				limit = {
					this != scope:jip_convert
					is_male = yes
					religion = religion:judaism_religion
				}
				order_by = {
					value = age
					if = {
						limit = { is_child_of = scope:jip_convert }
						add = 3000
					}
					else_if = {
						limit = { is_close_family_of = scope:jip_convert }
						add = 2000
					}
					else_if = {
						limit = { is_extended_family_of = scope:jip_convert }
						add = 1000
					}
				}
				save_scope_as = jip_new_head
			}
		}
		create_title_and_vassal_change = {
			type = stepped_down
			save_scope_as = jip_convert_change
			add_claim_on_loss = no
		}
		every_held_title = {
			change_title_holder_include_vassals = {
				holder = scope:jip_new_head
				change = scope:jip_convert_change
			}
		}
		resolve_title_and_vassal_change = scope:jip_convert_change
		# A landed heir keeps their own government and realm, the family title just joins their other titles
		scope:jip_new_head = {
			if = {
				limit = { is_landed = no }
				if = {
					limit = { NOT = { has_government = mishpacha_government } }
					change_government = mishpacha_government
				}
				jip_mishpacha_ensure_succession_law_effect = yes
				# Keep the family where it was and under the same liege
				if = {
					limit = { exists = scope:jip_convert_location }
					domicile ?= { move_domicile = scope:jip_convert_location }
				}
				if = {
					limit = {
						exists = scope:jip_convert_liege
						scope:jip_convert_liege != this
					}
					jip_mishpacha_change_liege_effect = { LIEGE = scope:jip_convert_liege }
				}
			}
		}
	}
	else = {
		every_held_title = {
			limit = { jip_is_mishpacha_family_title_trigger = yes }
			scope:jip_convert = { destroy_title = prev }
		}
		change_government = administrative_government
	}
}

# Gives a Mishpacha family head the Mishpacha realm succession law when they lack one.
# The engine picks it on its own at game start; this covers heads made by script and saves from before the law existed.
jip_mishpacha_ensure_succession_law_effect = {
	if = {
		limit = {
			has_government = mishpacha_government
			NOT = { has_realm_law = mishpacha_succession_law }
		}
		add_realm_law_skip_effects = mishpacha_succession_law
	}
}

# Gives a Mishpacha family head the prestige modifier matching their Villa tier, in place of the Villa's influence.
# Removes it from anyone who is no longer a Mishpacha family head.
# root = the character
jip_mishpacha_refresh_villa_modifier_effect = {
	if = {
		limit = { has_character_modifier = jip_mishpacha_villa_1 }
		remove_character_modifier = jip_mishpacha_villa_1
	}
	if = {
		limit = { has_character_modifier = jip_mishpacha_villa_2 }
		remove_character_modifier = jip_mishpacha_villa_2
	}
	if = {
		limit = { has_character_modifier = jip_mishpacha_villa_3 }
		remove_character_modifier = jip_mishpacha_villa_3
	}
	if = {
		limit = { has_character_modifier = jip_mishpacha_villa_4 }
		remove_character_modifier = jip_mishpacha_villa_4
	}
	if = {
		limit = { has_character_modifier = jip_mishpacha_villa_5 }
		remove_character_modifier = jip_mishpacha_villa_5
	}
	if = {
		limit = {
			has_government = mishpacha_government
			exists = domicile
		}
		domicile = { save_temporary_scope_as = jip_villa_domicile }
		if = {
			limit = { scope:jip_villa_domicile = { has_domicile_building_or_higher = estate_main_05 } }
			add_character_modifier = jip_mishpacha_villa_5
		}
		else_if = {
			limit = { scope:jip_villa_domicile = { has_domicile_building_or_higher = estate_main_04 } }
			add_character_modifier = jip_mishpacha_villa_4
		}
		else_if = {
			limit = { scope:jip_villa_domicile = { has_domicile_building_or_higher = estate_main_03 } }
			add_character_modifier = jip_mishpacha_villa_3
		}
		else_if = {
			limit = { scope:jip_villa_domicile = { has_domicile_building_or_higher = estate_main_02 } }
			add_character_modifier = jip_mishpacha_villa_2
		}

		}
	}

# Gives a Mishpacha family the Jewish courtiers its court lacks: a scholar, a man of business and a go-between when no
# adult Jew at court is good at learning, stewardship or diplomacy, and a cantor if that still leaves fewer than four.
# Templates in common/scripted_character_templates/jip_jewish_courtier_templates.txt. Used at game start.
# root = the family head
jip_mishpacha_generate_courtiers_effect = {
	jip_mishpacha_add_jewish_courtier_effect = { SKILL = learning TEMPLATE = jip_jewish_scholar_courtier_template }
	jip_mishpacha_add_jewish_courtier_effect = { SKILL = stewardship TEMPLATE = jip_jewish_steward_courtier_template }
	jip_mishpacha_add_jewish_courtier_effect = { SKILL = diplomacy TEMPLATE = jip_jewish_envoy_courtier_template }
	if = {
		limit = {
			NOT = {
				any_courtier = {
					count >= 4
					has_religion = religion:judaism_religion
					is_adult = yes
				}
			}
		}
		jip_mishpacha_create_jewish_courtier_effect = { TEMPLATE = jip_jewish_cantor_courtier_template }
	}
}

# A Jewish courtier from $TEMPLATE$, unless an adult Jew at court already has 10 or more $SKILL$
jip_mishpacha_add_jewish_courtier_effect = {
	if = {
		limit = {
			NOT = {
				any_courtier = {
					has_religion = religion:judaism_religion
					is_adult = yes
					$SKILL$ >= 10
				}
			}
		}
		jip_mishpacha_create_jewish_courtier_effect = { TEMPLATE = $TEMPLATE$ }
	}
}
