fix_kemet_title = {
	if = {
		limit = {
			title:e_kemet = {
				has_custom_title_name = no
			}
		}
		if = {
			limit = {
				OR = {
					root.culture = { has_name_list = name_list_ashkenazi }
					root.culture = { has_name_list = name_list_sephardi }
					root.culture = { has_name_list = name_list_radhanite }
					root.culture = { has_name_list = name_list_canaanite }
					root.culture = { has_name_list = name_list_assyrian }
					root.culture = { has_name_list = name_list_nubian }
					root.culture = { has_name_list = name_list_ancient_egyptian }
					root.culture = { has_name_list = name_list_greek }
					root.culture = { has_name_list = name_list_roman }
				}
			}
			title:e_kemet = {
				set_definitive_form = no
			}
		}
		else = {
			title:e_kemet = {
				set_definitive_form = yes 
			}
		}
	}
}

# A landed Mishpacha family takes on its liege's government, or feudal when independent
# or when the liege's government doesn't suit a landed vassal.
jip_mishpacha_adopt_landed_government_effect = {
	if = {
		limit = {
			is_independent_ruler = no
			liege ?= {
				NOR = {
					government_has_flag = government_is_mishpacha
					government_has_flag = government_is_theocracy
					government_has_flag = government_is_ecclesiastical
					government_has_flag = government_is_holy_order
					government_has_flag = government_is_mercenary
					government_has_flag = government_is_landless_adventurer
				}
			}
		}
		liege.government_type = { save_scope_as = jip_new_government }
		change_government = scope:jip_new_government
	}
	else = {
		change_government = feudal_government
	}
}
