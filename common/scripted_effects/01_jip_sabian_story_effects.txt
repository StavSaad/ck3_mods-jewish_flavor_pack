# The Return of the Moon God, the Sabian story (common/story_cycles/jip_story_sabian_revival.txt)

# Starts the story for the head of Thabit ibn Qurra's house (jip_sabian_story_on_actions.txt; no root needed). Thabit
# himself keeps the old gods in secret from the start, the first of the story's initiates (on_setup)
jip_sabian_story_start_effect = {
	title:d_jip_harrani.holder ?= {
		if = {
			limit = {
				jip_sabian_story_keeper_trigger = yes
				NOT = { jip_owns_sabian_story_trigger = yes }
			}
			if = {
				limit = {
					this = character:jip_sabian_thabit_ibn_qurra
					jip_sabian_initiate_candidate_trigger = yes
				}
				make_character_crypto_religionist_effect = { CRYPTO_RELIGION = faith:mesopotamian_pagan }
			}
			create_story = story_jip_sabian_revival
		}
	}
}

# Character scope: raises the standing of the Sabian community by $VALUE$ (1-3), if they own the story
jip_sabian_story_raise_standing_effect = {
	if = {
		limit = { jip_owns_sabian_story_trigger = yes }
		custom_tooltip = jip_sabian_standing_up_$VALUE$_tt
		hidden_effect = {
			random_owned_story = {
				type = story_jip_sabian_revival
				change_variable = {
					name = jip_sabian_standing
					add = $VALUE$
				}
				jip_sabian_story_check_standing_effect = yes
			}
		}
	}
}

# Character scope: lowers the standing of the Sabian community by $VALUE$ (1-3), if they own the story
jip_sabian_story_lower_standing_effect = {
	if = {
		limit = { jip_owns_sabian_story_trigger = yes }
		custom_tooltip = jip_sabian_standing_down_$VALUE$_tt
		hidden_effect = {
			random_owned_story = {
				type = story_jip_sabian_revival
				change_variable = {
					name = jip_sabian_standing
					subtract = $VALUE$
				}
				jip_sabian_story_check_standing_effect = yes
			}
		}
	}
}

# Story scope: keeps the standing within 0-10; at 10 in Act 1 the act ends
jip_sabian_story_check_standing_effect = {
	clamp_variable = {
		name = jip_sabian_standing
		min = 0
		max = 10
	}
	if = {
		limit = {
			var:jip_sabian_act = 1
			var:jip_sabian_standing >= 10
		}
		set_variable = {
			name = jip_sabian_act
			value = 2
		}
		story_owner = {
			trigger_event = {
				id = jip_sabian_story.1090
				days = 3
			}
		}
	}
}

# Story scope: the patron is the owner's liege, or the suzerain of a tributary family; from Act IV, when Harran is free,
# there is none
jip_sabian_story_update_patron_effect = {
	story_owner = {
		if = {
			limit = {
				is_independent_ruler = no
				prev.var:jip_sabian_act < 4
			}
			liege = { save_scope_as = jip_sabian_patron }
		}
		else_if = {
			limit = {
				exists = suzerain
				prev.var:jip_sabian_act < 4
			}
			suzerain = { save_scope_as = jip_sabian_patron }
		}
	}
	if = {
		limit = { exists = scope:jip_sabian_patron }
		set_variable = {
			name = jip_sabian_patron
			value = scope:jip_sabian_patron
		}
	}
	else = {
		remove_variable = jip_sabian_patron
	}
}

# Story scope: drops initiates who have died, left the house's court or given up the secret faith
jip_sabian_story_update_initiates_effect = {
	every_in_list = {
		variable = jip_sabian_initiates
		limit = {
			OR = {
				is_alive = no
				NOT = { secret_faith ?= faith:mesopotamian_pagan }
				AND = {
					this != scope:story.story_owner
					NOT = { is_courtier_of = scope:story.story_owner }
				}
			}
		}
		save_temporary_scope_as = jip_lapsed
		scope:story = {
			remove_list_variable = {
				name = jip_sabian_initiates
				target = scope:jip_lapsed
			}
		}
	}
}

# Character scope: makes them a secret believer in the old gods and adds them to the story's initiates; $HEAD$ = the story
# owner
jip_sabian_initiate_effect = {
	make_character_crypto_religionist_effect = { CRYPTO_RELIGION = faith:mesopotamian_pagan }
	save_temporary_scope_as = jip_new_initiate
	$HEAD$ = {
		random_owned_story = {
			type = story_jip_sabian_revival
			add_to_variable_list = {
				name = jip_sabian_initiates
				target = scope:jip_new_initiate
			}
		}
	}
}

# Character scope: an initiate who recants, giving up the secret faith
jip_sabian_recant_effect = {
	if = {
		limit = { secret_faith ?= faith:mesopotamian_pagan }
		random_secret = {
			type = secret_crypto_religionist
			remove_secret = yes
		}
		remove_character_secret_faith = yes
	}
}

# root = the family head finishing a contract (jip_contract_finish_effect): a translation, horoscope or cure for a Muslim
# patron raises the Sabians' standing
jip_sabian_story_contract_effect = {
	if = {
		limit = {
			jip_owns_sabian_story_trigger = yes
			OR = {
				flag:$CONTRACT$ = flag:translate
				flag:$CONTRACT$ = flag:horoscope
				flag:$CONTRACT$ = flag:physician
			}
			scope:task_contract.task_contract_employer ?= { religion = religion:islam_religion }
		}
		if = {
			limit = { jip_contract_outcome_is_trigger = { OUTCOME = critical } }
			jip_sabian_story_raise_standing_effect = { VALUE = 2 }
		}
		else_if = {
			limit = { jip_contract_outcome_is_trigger = { OUTCOME = success } }
			jip_sabian_story_raise_standing_effect = { VALUE = 1 }
		}
	}
}

### Act II: the Return to Harran

# Character scope: takes Harran (with the old count's baronies in it, and its vassals) as a $TYPE$ title change, then
# becomes the vassal of scope:jip_harran_liege if that exists, else stays where they are
jip_sabian_take_harran_effect = {
	save_scope_as = jip_harran_taker
	title:c_harran.holder = { save_scope_as = jip_harran_loser }
	create_title_and_vassal_change = {
		type = $TYPE$
		save_scope_as = jip_harran_change
		add_claim_on_loss = no
	}
	title:c_harran = {
		every_in_de_jure_hierarchy = {
			limit = {
				tier = tier_barony
				holder ?= scope:jip_harran_loser
			}
			change_title_holder = {
				holder = scope:jip_harran_taker
				change = scope:jip_harran_change
			}
		}
		change_title_holder_include_vassals = {
			holder = scope:jip_harran_taker
			change = scope:jip_harran_change
		}
	}
	if = {
		limit = {
			exists = scope:jip_harran_liege
			scope:jip_harran_liege != scope:jip_harran_taker
			NOT = { liege ?= scope:jip_harran_liege }
		}
		change_liege = {
			liege = scope:jip_harran_liege
			change = scope:jip_harran_change
		}
	}
	resolve_title_and_vassal_change = scope:jip_harran_change
}

# Character scope (Harran's seller): saves who the buyer will serve: the seller if they keep a duchy or more, else the
# seller's liege; an independent count's buyer becomes independent too
jip_sabian_harran_sale_liege_effect = {
	if = {
		limit = {
			any_held_title = {
				tier >= tier_duchy
				is_landless_type_title = no
			}
		}
		save_scope_as = jip_harran_liege
	}
	else_if = {
		limit = { is_independent_ruler = no }
		liege = { save_scope_as = jip_harran_liege }
	}
}

# Character scope (the story owner): raises the Harranians: a pressed claim on Harran, a claim war on its count, and an army
# of Harranians that lasts as long as the war
jip_sabian_raise_the_harranians_effect = {
	title:c_harran.holder = { save_scope_as = jip_harran_loser }
	add_pressed_claim = title:c_harran
	start_war = {
		casus_belli = claim_cb
		target = scope:jip_harran_loser
		claimant = root
		target_title = title:c_harran
	}
	random_character_war = {
		limit = {
			primary_attacker = root
			primary_defender = scope:jip_harran_loser
		}
		save_scope_as = jip_harran_war
	}
	spawn_army = {
		levies = jip_sabian_harranian_levies_value
		location = title:c_harran.title_province
		inheritable = no
		uses_supply = yes
		war = scope:jip_harran_war
		name = jip_sabian_harranian_army
	}
}

# The story's act moves on; Act II ends when the owner holds Harran (jip_sabian_story_harran_gained_on_action)
jip_sabian_story_set_act_effect = {
	random_owned_story = {
		type = story_jip_sabian_revival
		set_variable = {
			name = jip_sabian_act
			value = $ACT$
		}
	}
}

### Act III: Throw Off the Caliph

# Character scope (the liege): lets scope:jip_rebel go: independent if the liege is a top liege, else the liege's liege's
# vassal (as vanilla's independence war does), with a five-year truce
jip_sabian_grant_freedom_effect = {
	create_title_and_vassal_change = {
		type = independency
		save_scope_as = change
	}
	change_liege_or_become_independent = {
		CHANGE = scope:change
		VASSAL = scope:jip_rebel
	}
	resolve_title_and_vassal_change = scope:change
	add_truce_both_ways = {
		character = scope:jip_rebel
		days = 1825
		name = jip_sabian_freedom_truce
	}
}

# Character scope (the story owner): fights for Harran's freedom: vanilla's independence war against their liege, with an
# army of Harranians that serves until it ends
jip_sabian_fight_for_freedom_effect = {
	liege = { save_scope_as = jip_harran_overlord }
	start_war = {
		casus_belli = independence_war
		target = scope:jip_harran_overlord
	}
	random_character_war = {
		limit = {
			primary_attacker = root
			primary_defender = scope:jip_harran_overlord
		}
		save_scope_as = jip_freedom_war
	}
	spawn_army = {
		levies = jip_sabian_freedom_levies_value
		location = title:c_harran.title_province
		inheritable = no
		uses_supply = yes
		war = scope:jip_freedom_war
		name = jip_sabian_harranian_army
	}
}

### Act IV: Restore the Old Gods

# Character scope (the story owner): the mysteries unveiled. The owner proclaims the old gods if they haven't (ruler, capital,
# Harran: jip_proclaim_old_gods_effect), the initiates give up their secret and take the owner's rite, and so do the Sabians
# of the court, the realm's Sabian vassals and its Sabian counties. Muslim neighbours resent it, and their faiths grow
# fervent
jip_sabian_unveil_the_mysteries_effect = {
	if = {
		limit = { NOT = { faith = faith:mesopotamian_pagan } }
		jip_proclaim_old_gods_effect = yes
	}
	rite = { save_scope_as = jip_old_gods_rite }
	random_owned_story = {
		type = story_jip_sabian_revival
		every_in_list = {
			variable = jip_sabian_initiates
			limit = { is_alive = yes }
			jip_sabian_recant_effect = yes
			if = {
				limit = { NOT = { faith = faith:mesopotamian_pagan } }
				set_character_rite = scope:jip_old_gods_rite
			}
		}
		set_variable = jip_sabian_unveiled
	}
	every_courtier = {
		limit = { rite = rite:sabianism }
		set_character_rite = scope:jip_old_gods_rite
	}
	every_vassal_or_below = {
		limit = { rite = rite:sabianism }
		set_character_rite = scope:jip_old_gods_rite
	}
	every_sub_realm_county = {
		limit = { rite = rite:sabianism }
		set_county_rite = scope:jip_old_gods_rite
	}
	every_neighboring_and_across_water_top_liege_realm_owner = {
		limit = { religion = religion:islam_religion }
		add_opinion = {
			target = root
			modifier = jip_restored_old_gods_opinion
		}
		faith = { add_to_temporary_list = jip_angered_faiths }
	}
	every_in_list = {
		list = jip_angered_faiths
		change_fervor = {
			value = slight_fervor_gain
			desc = fervor_gain_jip_old_gods_restored
		}
	}
}

# Character scope (the story owner): a High Priest of Sin, of the owner's rite, becomes their court chaplain
jip_sabian_name_high_priest_effect = {
	create_character = {
		location = root.capital_province
		culture = root.culture
		faith = root.faith
		gender_female_chance = 0
		age = { 40 60 }
		learning = { 12 18 }
		trait = education_learning_3
		trait = zealous
		employer = root
		save_scope_as = jip_high_priest
	}
	scope:jip_high_priest = {
		set_character_rite = root.rite
		give_nickname = nick_jip_high_priest_of_sin
	}
	assign_councillor_type = {
		type = councillor_court_chaplain
		target = scope:jip_high_priest
	}
}

# Character scope (a feast's host): the New Year kept (Akitu, jip_feast.1004, or the Feast of Nisan for the old faith,
# jip_feast.2004). After the unveiling, the first one ends Act IV
jip_sabian_story_new_year_kept_effect = {
	if = {
		limit = {
			any_character_artifact = { has_variable = jip_old_god_statue }
		}
		custom_tooltip = jip_sabian_statues_at_new_year_tt
		hidden_effect = { jip_sabian_statues_at_new_year_effect = yes }
	}
	if = {
		limit = {
			jip_sabian_story_in_act_trigger = { ACT = 4 }
			jip_sabian_story_unveiled_trigger = yes
		}
		jip_sabian_story_set_act_effect = { ACT = 5 }
		trigger_event = {
			id = jip_sabian_story.4090
			days = 3
		}
	}
}

### Act V: Reclaim the Land

# Character scope (the story owner): a pressed claim on the first cult city, in the order Harran, Mosul (Ashur and Hatra),
# Nineveh, Arbela, Palmyra, that lies outside the realm and isn't of the old faith
jip_sabian_claim_cult_city_effect = {
	if = {
		limit = { title:c_harran = { jip_sabian_cult_city_to_claim_trigger = { OWNER = root } } }
		title:c_harran = { save_scope_as = jip_cult_city }
	}
	else_if = {
		limit = { title:c_mosul = { jip_sabian_cult_city_to_claim_trigger = { OWNER = root } } }
		title:c_mosul = { save_scope_as = jip_cult_city }
	}
	else_if = {
		limit = { title:c_ninive = { jip_sabian_cult_city_to_claim_trigger = { OWNER = root } } }
		title:c_ninive = { save_scope_as = jip_cult_city }
	}
	else_if = {
		limit = { title:c_irbil = { jip_sabian_cult_city_to_claim_trigger = { OWNER = root } } }
		title:c_irbil = { save_scope_as = jip_cult_city }
	}
	else_if = {
		limit = { title:c_palmyra = { jip_sabian_cult_city_to_claim_trigger = { OWNER = root } } }
		title:c_palmyra = { save_scope_as = jip_cult_city }
	}
	if = {
		limit = { exists = scope:jip_cult_city }
		add_pressed_claim = scope:jip_cult_city
	}
}

# Character scope: the statue of $GOD$ as a court artifact (a holy relic of the old faith)
jip_sabian_create_god_statue_effect = {
	create_artifact = {
		name = jip_artifact_statue_$GOD$_name
		description = jip_artifact_statue_$GOD$_desc
		visuals = statue_mena
		type = pedestal
		template = jip_old_god_statue_template
		modifier = jip_statue_of_$GOD$_modifier
		wealth = 60
		quality = 80
		decaying = no
		save_scope_as = jip_statue
	}
	scope:jip_statue = { set_variable = jip_old_god_statue }
	add_to_global_variable_list = {
		name = jip_old_gods_home
		target = flag:$GOD$
	}
}

# Character scope (the story owner): rolls the search for scope:jip_sought_god with scope:jip_search_skill (learning or
# intrigue): the statue is found (jip_sabian_story.5011) or the trail goes cold (.5012), four to eight months on

	random_list = {
		0 = {
			modifier = {
				add = jip_sabian_statue_search_chance_value
			}

				id = jip_sabian_story.5011
				months = { 4 8 }
			}
		}
		0 = {
			modifier = {
				add = {
					value = 100
					subtract = jip_sabian_statue_search_chance_value
				}
			}
			trigger_event = {
				id = jip_sabian_story.5012
				months = { 4 8 }
			}
		}
	}
}
