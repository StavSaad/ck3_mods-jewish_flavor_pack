# This file contains the setup effects for the mod's tributaries.
# Sections are organized by top-suzerain title.
# Note that existing vassals (as established in title history) must be released and recreated into tributaries before
# they can have their own tributaries.

# Mishpacha counterpart of vanilla's break_subject_contract_and_establish_tributary_effect,
# using the tributary_mishpacha contract group instead of tributary_settled. Call in the tributary's scope.
# Non-Mishpacha tributaries fall back to the vanilla effect.
# $TRIBUTARY$ = the Mishpacha family head
# $SUZERAIN$ = their new suzerain
jip_establish_mishpacha_tributary_effect = {
	if = {
		limit = {
			$TRIBUTARY$ = { NOT = { has_government = mishpacha_government } }
		}
		break_subject_contract_and_establish_tributary_effect = { SUZERAIN = $SUZERAIN$ TRIBUTARY = $TRIBUTARY$ }
	}
	else = {
		jip_establish_mishpacha_tributary_inner_effect = { SUZERAIN = $SUZERAIN$ TRIBUTARY = $TRIBUTARY$ }
	}
}

jip_establish_mishpacha_tributary_inner_effect = {
	if = {
		limit = { liege != this }
		create_title_and_vassal_change = {
			type = independency
			save_scope_as = change
			add_claim_on_loss = no
		}
		becomes_independent = {
			change = scope:change
		}
		resolve_title_and_vassal_change = scope:change
	}
	$TRIBUTARY$ = {
		start_tributary = {
			contract_group = tributary_mishpacha
			suzerain = $SUZERAIN$
		}
		if = {
			limit = { is_tributary = yes } # suppress error logging
			hidden_effect = {
				tributary_contract_set_obligation_level = { type = default_tributary_taxes level = 2 }
				tributary_contract_set_obligation_level = { type = suzerain_war_participation_guarantee level = 1 }
			}
		}
	}
	setup_tributary_truces_effect = { SUZERAIN = $SUZERAIN$ TRIBUTARY = $TRIBUTARY$ }
}

# Invoked from jip_tributary_effect_setup in on_game_start, so vanilla's setup_tributaries_effect stays untouched.
jip_tributary_setup_effect = {
	if = {
		limit = { current_date >= 1178.1.1 }
		jip_1178_tributary_setup_effect = yes
	}
	else_if = {
		limit = { current_date >= 1066.1.1 }
		jip_1066_tributary_setup_effect = yes
	}
	else_if = {
		limit = { current_date >= 867.1.1 }
		jip_867_tributary_setup_effect = yes
	}
}

jip_867_tributary_setup_effect = {
	character:34014 ?= { # Abbasid Caliph
		save_scope_as = suzerain
		character:rabbinic_exilarchate_pinchas_ben_abdimi ?= {
			jip_establish_mishpacha_tributary_effect = { SUZERAIN = scope:suzerain TRIBUTARY = this }
		}
		character:rabbinic_exilarchate_hasdai_ii ?= {
			jip_establish_mishpacha_tributary_effect = { SUZERAIN = scope:suzerain TRIBUTARY = this }
		}
