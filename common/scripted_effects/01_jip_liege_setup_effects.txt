# This file contains the game start liege setup for the mod's Mishpacha families.
# Sections are organized by liege.

# Puts a landless Mishpacha family head under $LIEGE$, the ruler of the realm their estate lies in. Call in the family head's scope.
# Family titles are duchy tier, so only a king or emperor can take them in as vassals; a duke or lesser ruler takes tribute
# from them instead (vanilla's settled tributary contract). Any tie to a former liege or suzerain ends. Their family title
# survives independence because our copy of yearly_on_actions.txt spares Mishpacha families from vanilla's cleanup.
# Ranks are compared with tier_duchy, the family title's tier, not the head's highest_held_title_tier, which may not count
# a family title; if it doesn't, a duke passes as a valid liege, the engine refuses the fealty and the family keeps its old liege.
jip_mishpacha_change_liege_effect = {
	$LIEGE$ = { save_scope_as = jip_new_liege }
	if = {
		limit = {
			scope:jip_new_liege != this
			scope:jip_new_liege.highest_held_title_tier <= tier_duchy
		}
		jip_mishpacha_become_tributary_effect = yes
	}
	# Already their vassal: nothing to change (otherwise the fallback below would make them independent)
	else_if = {
		limit = {
			NOT = { liege ?= scope:jip_new_liege }
		}
		if = {
			limit = { is_tributary = yes }
			end_tributary = yes
		}
		if = {
			limit = {
				scope:jip_new_liege != this
				scope:jip_new_liege.highest_held_title_tier > tier_duchy
			}
			create_title_and_vassal_change = {
				type = swear_fealty
				save_scope_as = jip_liege_change
			}
			change_liege = {
				liege = scope:jip_new_liege
				change = scope:jip_liege_change
			}
			resolve_title_and_vassal_change = scope:jip_liege_change
			# Dhimmi status: under a liege who taxes nonbelievers, the charter starts with jizya (same check as the jizya option's is_valid)
			if = {
				limit = {
					is_independent_ruler = no
					liege = scope:jip_new_liege
					faith != scope:jip_new_liege.faith
					scope:jip_new_liege = {
						OR = {
							rite = { rite_has_tenet = tenet_tax_nonbelievers }
							faith = { has_doctrine = special_doctrine_jizya }
						}
					}
				}
				vassal_contract_set_obligation_level = { type = jizya_special_rights level = 1 }
			}
		}
		else_if = {
			limit = { is_independent_ruler = no }
			create_title_and_vassal_change = {
				type = independency
				save_scope_as = jip_liege_change
				add_claim_on_loss = no
			}
			becomes_independent = {
				change = scope:jip_liege_change
			}
			resolve_title_and_vassal_change = scope:jip_liege_change
		}
	}
}

# Part of jip_mishpacha_change_liege_effect: the family leaves any liege or other suzerain and pays tribute to
# scope:jip_new_liege, the duke or lesser ruler whose land their estate lies in
jip_mishpacha_become_tributary_effect = {
	if = {
		limit = {
			NOT = { suzerain ?= scope:jip_new_liege }
		}
		if = {
			limit = { is_tributary = yes }
			end_tributary = yes
		}
		# A vassal can't become a tributary: leave the old liege first, as vanilla's tributary setup does
		if = {
			limit = { liege != this }
			create_title_and_vassal_change = {
				type = independency
				save_scope_as = jip_liege_change
				add_claim_on_loss = no
			}
			becomes_independent = {
				change = scope:jip_liege_change
			}
			resolve_title_and_vassal_change = scope:jip_liege_change
		}
		start_tributary = {
			contract_group = tributary_settled
			suzerain = scope:jip_new_liege
		}
		# The standing vanilla gives a new tributary (start_tributary_interaction_effect)
		if = {
			limit = { is_tributary = yes }
			add_subject_standing = 10
		}
	}
}

# Invoked from jip_liege_effect_setup in on_game_start.
jip_liege_setup_effect = {
	if = {
		limit = { current_date >= 1178.1.1 }
		jip_1178_liege_setup_effect = yes
	}
	else_if = {
		limit = { current_date >= 1066.1.1 }
		jip_1066_liege_setup_effect = yes
	}
	else_if = {
		limit = { current_date >= 867.1.1 }
		jip_867_liege_setup_effect = yes
	}
}

jip_867_liege_setup_effect = {
	character:34014 ?= { # Abbasid Caliph
		save_scope_as = jip_setup_liege
		character:rabbinic_exilarchate_pinchas_ben_abdimi ?= {
			jip_mishpacha_change_liege_effect = { LIEGE = scope:jip_setup_liege }
		}
		character:rabbinic_exilarchate_hasdai_ii ?= {
			jip_mishpacha_change_liege_effect = { LIEGE = scope:jip_setup_liege }
