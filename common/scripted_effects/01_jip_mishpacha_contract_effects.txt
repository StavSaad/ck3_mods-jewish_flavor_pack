# Mishpacha contracts (common/task_contracts/jip_mishpacha_contracts.txt)

# Adds fresh contracts around the estate; offers not taken drop off after about a year (jip_mishpacha_contract_keep_trigger).
# Runs yearly, when the estate arrives somewhere new, when a new head takes over and at game start.
# root = the family head
jip_mishpacha_refresh_contracts_effect = {
	if = {
		limit = {
			jip_mishpacha_contract_owner_trigger = yes
			NOT = {
				any_character_task_contract = { count >= 6 }
			}
		}
		domicile.domicile_location = { save_scope_as = jip_contract_area }
		hidden_effect = {
			populate_task_contracts_for_area = {
				location = scope:jip_contract_area
				amount = jip_contract_refresh_amount_value
				group = { jip_mishpacha_contracts_group }
			}
		}
	}
}

# AI family heads: a yearly hidden roll in place of contracts. The work goes by their non-martial skills, and the better
# the skill, the better the odds and the pay. root = the family head
jip_mishpacha_ai_contract_effect = {
	random = {
		chance = 50
		random_list = {
			1 = {
				modifier = { add = stewardship }
				save_scope_value_as = { name = jip_ai_skill value = stewardship }
			}
			1 = {
				modifier = { add = diplomacy }
				save_scope_value_as = { name = jip_ai_skill value = diplomacy }
			}
			1 = {
				modifier = { add = learning }
				save_scope_value_as = { name = jip_ai_skill value = learning }
			}
			1 = {
				modifier = { add = intrigue }
				save_scope_value_as = { name = jip_ai_skill value = intrigue }
			}
		}
		random = {
			chance = jip_contract_ai_chance_value
			add_gold = jip_contract_ai_gold_value
			add_prestige = minor_prestige_gain
		}
	}
}

# Shared contract on_create: root = the family head, scope:contract = the new contract
jip_mishpacha_contract_on_create_effect = {
	scope:contract = {
		save_scope_as = task_contract
		task_contract_employer = { save_scope_as = employer }
		set_variable = {
			name = jip_contract_fresh
			value = yes
			days = 395
		}
		set_variable = {
			name = jip_contract_gold
			value = jip_contract_gold_value
		}
		set_variable = {
			name = jip_contract_stake
			value = jip_contract_stake_value
		}
	}
}

# Shared contract on_accepted: the work is done from home through letters and agents, and comes to a head a few months on
# root = the contract, $EVENT$ = the event that resolves it, $ACCEPT$ = the event that says so at once (.00X2)
jip_mishpacha_contract_on_accepted_effect = {
	task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_accept_contract" }
	save_scope_as = task_contract
	task_contract_taker ?= {
		trigger_event = $ACCEPT$
		trigger_event = {
			id = $EVENT$
			months = { 2 4 }
		}
	}
}

# Scheme contracts start at once: the approach event (.00X0) starts the contract scheme
# root = the contract, $EVENT$ = the approach event
jip_mishpacha_contract_on_accepted_scheme_effect = {
	task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_accept_contract" }
	save_scope_as = task_contract
	task_contract_taker ?= { trigger_event = $EVENT$ }
}

# Pays the stake out of the family's purse when a loan or caravan contract is taken. root = the contract
jip_mishpacha_contract_pay_stake_effect = {
	task_contract_taker ?= {
		remove_short_term_gold = scope:task_contract.var:jip_contract_stake
	}
	set_variable = {
		name = jip_contract_stake_paid
		value = yes
	}
}

# Shared contract on_invalidated: tells the taker, and hands back a stake that was paid. root = the contract
jip_mishpacha_contract_on_invalidated_effect = {
	save_scope_as = task_contract
	scheme ?= { end_scheme = yes }
	task_contract_taker ?= {
		if = {
			limit = { exists = scope:task_contract.var:jip_contract_stake_paid }
			add_gold = scope:task_contract.var:jip_contract_stake
		}
		send_interface_message = {
			type = event_generic_bad_with_text
			title = jip_mishpacha_contract_invalidated_title
			desc = jip_mishpacha_contract_invalidated_desc
			left_icon = this
		}
	}
}

# Saves the scopes the contract events use. root = the family head, scope:task_contract = the contract
jip_mishpacha_contract_event_scopes_effect = {
	scope:task_contract = {
		task_contract_employer ?= { save_scope_as = employer }
		var:jip_contract_target ?= { save_scope_as = jip_contract_target }
	}
}

# The family head picks an approach: rolls its outcome in secret and sends the result event a few days on, where the
# contract is completed. The tooltip shows the skill tested and the odds of each outcome.
# $SKILL$ = the skill tested (jip_contract_<skill>_chance_value), $BONUS$ = the approach's own bonus or malus to it,
# $APPROACH$ = flag for the result event and jip_contract_<contract>_approach_effect, $EVENT$ = the result event
# root = the family head, scope:task_contract = the contract
jip_contract_choose_approach_effect = {
	save_scope_value_as = {
		name = jip_contract_chance
		value = {
			value = jip_contract_$SKILL$_chance_value
			add = $BONUS$
			min = 5
			max = 95
		}
	}
	custom_tooltip = jip_contract_tests_$SKILL$_tt
	scope:task_contract = {
		set_variable = {
			name = jip_contract_approach
			value = flag:$APPROACH$
		}
	}
	random_list = {
		0 = {
			modifier = {
				add = {
					value = scope:jip_contract_chance
					multiply = 0.3
				}
			}
			desc = jip_contract_outcome_critical_tt
			hidden_effect = { jip_contract_set_outcome_effect = { OUTCOME = critical } }
		}
		0 = {
			modifier = {
				add = {
					value = scope:jip_contract_chance
					multiply = 0.7
				}
			}
			desc = jip_contract_outcome_success_tt
			hidden_effect = { jip_contract_set_outcome_effect = { OUTCOME = success } }
		}
		0 = {
			modifier = {
				add = {
					value = 100
					subtract = scope:jip_contract_chance
				}
			}
			desc = jip_contract_outcome_failure_tt
			hidden_effect = { jip_contract_set_outcome_effect = { OUTCOME = failure } }
		}
	}
	hidden_effect = {
		trigger_event = {
			id = $EVENT$
			days = { 3 7 }
		}
	}
}

### Contract schemes (common/schemes/scheme_types/jip_mishpacha_contract_schemes.txt), following vanilla's laamp contract schemes

# A phase is done, or the scheme is clicked in the HUD: vanilla's preparations window, which executes into $EVENT$
# this = the scheme
jip_contract_scheme_prep_effect = {
	save_scope_as = scheme
	scope:scheme.task_contract ?= { save_scope_as = task_contract }
	save_scope_value_as = {
		name = follow_up_event
		value = event_id:$EVENT$
	}
	if = {
		limit = {
			NOT = { exists = scope:suppress_next_event }
		}
		scheme_owner = { trigger_event = scheme_critical_moments.0002 }
	}
}

# The scheme can't go on: the contract falls through with it (which refunds a stake). this = the scheme
jip_contract_scheme_invalidated_effect = {
	save_scope_as = scheme
	scheme_owner = {
		send_interface_message = {
			title = scheme_invalidated_title
			type = diplo_contract_scheme_invalidated_message
			right_icon = scope:scheme.scheme_target_character
			show_as_tooltip = {
				scope:scheme = { end_scheme = yes }
			}
		}
	}
	scope:scheme.task_contract ?= { invalidate_contract = yes }
}

# The scheme is executed: its success chance decides, and a success is exceptional one time in three, as in vanilla's
# roll_scheme_success_or_failure_effect. root = the family head
jip_contract_roll_scheme_effect = {
	random = {
		chance = scope:scheme.scheme_success_chance
		save_scope_value_as = {
			name = jip_scheme_successful
			value = yes
		}
	}
	if = {
		limit = { exists = scope:jip_scheme_successful }
		random_list = {
			67 = { jip_contract_set_outcome_effect = { OUTCOME = success } }
			33 = { jip_contract_set_outcome_effect = { OUTCOME = critical } }
		}
		trigger_event = $SUCCESS_EVENT$
	}
	else = {
		jip_contract_set_outcome_effect = { OUTCOME = failure }
		trigger_event = $FAILURE_EVENT$
	}
}


jip_contract_set_outcome_effect = {
	scope:task_contract = {
		set_variable = {
			name = jip_contract_outcome
			value = flag:$OUTCOME$
		}
	}
}

# The result event's choice: the approach's own consequences, then the contract's reward for the outcome rolled.
# $CONTRACT$ = the contract key without jip_contract_ (its jip_contract_<contract>_approach_effect), root = the family head
jip_contract_finish_effect = {
	jip_contract_$CONTRACT$_approach_effect = yes
	if = {
		limit = { jip_contract_outcome_is_trigger = { OUTCOME = critical } }
		scope:task_contract = { complete_task_contract = success_critical }
	}
	else_if = {
		limit = { jip_contract_outcome_is_trigger = { OUTCOME = success } }
		scope:task_contract = { complete_task_contract = success_standard }
	}
	else = {
		scope:task_contract = { complete_task_contract = failure_standard }
	}
}

### Approach consequences, on top of the contract's reward (root = the family head)

# Extra gold: $SHARE$ times the contract's gold, or of its stake
jip_contract_extra_gold_effect = {
	add_gold = {
		value = scope:task_contract.var:jip_contract_gold
		multiply = $SHARE$
	}
}

jip_contract_extra_stake_effect = {
	add_gold = {
		value = scope:task_contract.var:jip_contract_stake
		multiply = $SHARE$
	}
}

# Less gold: $SHARE$ (positive) times the contract's gold, or of its stake, is held back from the reward
jip_contract_less_gold_effect = {
	remove_short_term_gold = {
		value = scope:task_contract.var:jip_contract_gold
		multiply = $SHARE$
	}
}

jip_contract_less_stake_effect = {
	remove_short_term_gold = {
		value = scope:task_contract.var:jip_contract_stake
		multiply = $SHARE$
	}
}

# $WHO$'s opinion of the family head changes by $OPINION$ with the $TYPE$ modifier
jip_contract_extra_opinion_effect = {
	$WHO$ ?= {
		add_opinion = {
			target = root
			modifier = $TYPE$
			opinion = $OPINION$
		}
	}
}

jip_contract_loan_approach_effect = {
	# Courtesy: the friendship outlives the debt, paid or not
	if = {
		limit = { jip_contract_approach_is_trigger = { APPROACH = courtesy } }
		jip_contract_extra_opinion_effect = { WHO = scope:employer TYPE = grateful_opinion OPINION = 15 }
	}
	# Rents and goods: sold on at a profit, but the debtor resents what was taken from him
	else_if = {
		limit = { jip_contract_approach_is_trigger = { APPROACH = goods } }
		if = {
			limit = { jip_contract_succeeded_trigger = yes }
			jip_contract_extra_stake_effect = { SHARE = 0.25 }
		}
		jip_contract_extra_opinion_effect = { WHO = scope:employer TYPE = disappointed_opinion OPINION = -15 }
	}
	# Calling in his other debts: a hold over him if it works, an enemy if it doesn't
	else_if = {
		limit = { jip_contract_approach_is_trigger = { APPROACH = pressure } }
		if = {
			limit = { jip_contract_succeeded_trigger = yes }
			add_hook_if_possible_default_length_effect = {
				TARGET = scope:employer
				TYPE = contact_list_weak_hook
			}
		}
		else = {
			jip_contract_extra_opinion_effect = { WHO = scope:employer TYPE = disappointed_opinion OPINION = -30 }
		}
	}
}
