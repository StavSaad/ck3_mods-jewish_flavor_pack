# Traditions that unlock the mod's cultural men-at-arms (common/men_at_arms_types/jip_cultural_maa_types.txt), built
# like vanilla's in 00_maa_traditions.txt, each open to one heritage:
# - Syriac: Aramean (syriac) starts with the Palmyrene archers, Assyrian with the clibanarii. Neither costs extra for a
#   Bellicose (Aramean) or Communal (Assyrian) culture
# - Israelite: Edomite starts with the Bowmen of Seir (in place of vanilla's Desert Ribat)
# - Phoenician: Phoenician and Carthaginian start with the Sacred Band
# - Egyptian: Kemi (copt) and Ancient Egyptian start with Archers of the Two Lands, vanilla's Nile Archers
# Forming Hebrew keeps them (handle_maa_traditions_for_culture in 01_jip_decisions_effects.txt)

tradition_jip_palmyrene_archery = {
	category = regional

	layers = {
		0 = martial
		1 = mena
		4 = desert.dds
	}

	is_shown = {
		has_cultural_pillar = heritage_syriac
	}
	can_pick = {
		has_cultural_pillar = heritage_syriac
	}

	parameters = {
		unlock_maa_jip_palmyrene_archers = yes
	}
	character_modifier = {
		archers_damage_mult = 0.1
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_bellicose = { is_in_list = traits }
						culture_pillar:ethos_communal = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_bellicose_or_communal_desc
				}
			}
			if = {
				limit = {
					NOT = {
						scope:character = {
							number_maa_soldiers_of_base_type = {
								type = archers
								value >= 800
							}
						}
					}
				}
				add = {
					value = tradition_unfulfilled_criteria_penalty
					desc = culture_head_archers_desc
				}
			}

			if = {
				limit = {
					culture_tradition_reduction_trigger = { TRADITION = tradition_jip_palmyrene_archery }
				}
				multiply = {
					value = 0.5
					desc = inspired_by_culture_desc
				}
			}

			multiply = tradition_replacement_cost_if_relevant
		}
	}

	ai_will_do = {
		value = 400
		if = {
			limit = {
				NOT = {
					scope:character = {
						any_sub_realm_county = {
							culture = scope:character.culture
							county_has_province_with_terrain = {
								terrain = drylands
								terrain = desert
							}
						}
					}
				}
			}
			multiply = 0
		}
	}
}
