# An independent Sabian ruler who holds Harran rebuilds the temple Nabonidus once restored. Its completion opens the
# Mesopotamian pagan faith to conversion and offers the founder the choice to proclaim the old gods (jip_sabian.0001).
# Independent only: a project belongs to the top liege of its province, so under the caliph the caliph would own it
jip_gp_temple_of_sin = {
	icon = { reference = @jip_temple_of_sin_icon }
	illustration = { reference = @jip_temple_of_sin_illustration }

	name = {
		desc = great_project_type_jip_gp_temple_of_sin
	}

	is_shown = {
		OR = {
			rite = rite:sabianism
			faith = faith:mesopotamian_pagan
		}
		is_independent_ruler = yes

		scope:founder = {
			OR = {
				rite = rite:sabianism
				faith = faith:mesopotamian_pagan

#	jip_gp_temple_of_ishtar - the Temple of Ishtar at Arbela
#	jip_gp_temple_of_marduk - the Esagila, the temple of Marduk at Babylon
#	jip_gp_temple_of_melqart - the Temple of Melqart at Tyre, for Canaanism
#	jip_gp_temple_of_tanit - the Temple of Tanit at Carthage, for Canaanism
#	jip_gp_temple_of_eshmun - the Temple of Eshmun at Sidon, for Canaanism
#	jip_gp_temple_of_baalat_gebal - the Temple of Baalat Gebal at Byblos, for Canaanism
#	jip_gp_temple_of_baal - the Temple of Baal at Baalbek, for Canaanism
#	jip_gp_temple_of_astarte - the Temple of Astarte at Kition, for Canaanism

#################
