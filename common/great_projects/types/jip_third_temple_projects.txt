#################
#	The Third Temple as two great projects (0.16 plan, section 7; vanilla's system: _great_project_types.info)
#	jip_gp_clear_the_temple_mount - Clear the Temple Mount: the Dome of the Rock comes down (while it stands)
#	jip_gp_rebuild_the_temple - Rebuild the Temple: temple_in_jerusalem_01 (once the Mount is clear and no Temple stands)
#	Effects in common/scripted_effects/01_jip_third_temple_effects.txt
#################

@jip_clear_the_temple_mount_icon = "gfx/interface/icons/great_projects/_default.dds" # placeholder until its art
@jip_clear_the_temple_mount_illustration = "gfx/interface/illustrations/event_scenes/pam_cathedral_construction.dds" # placeholder
@jip_rebuild_the_temple_icon = "gfx/interface/icons/great_projects/_default.dds" # placeholder until its art
@jip_rebuild_the_temple_illustration = "gfx/interface/illustrations/event_scenes/pam_cathedral_construction.dds" # placeholder

# Stage 1. The old decision's founder: an independent Jewish ruler of Israelite heritage who holds Jerusalem and
# completely controls Israel, at piety level 4. While the work lasts every Muslim ruler resents the founder; on
# completion the Dome of the Rock is gone, Jerusalem is no longer an Islamic holy site, and jip_hebrew.0041 fires.
jip_gp_clear_the_temple_mount = {
	icon = { reference = @jip_clear_the_temple_mount_icon }
	illustration = { reference = @jip_clear_the_temple_mount_illustration }

	name = {
		desc = great_project_type_jip_gp_clear_the_temple_mount
	}

	is_shown = {
		culture = { has_cultural_pillar = heritage_israelite }
		has_religion = religion:judaism_religion
		is_independent_ruler = yes
		is_landed = yes
		NOT = { is_planning_great_project = jip_gp_clear_the_temple_mount }
		title:c_jerusalem.title_province = {
			has_building = dome_of_the_rock_01
			NOT = { has_building = temple_in_jerusalem_01 }
		}
	}

	province_filter = landed_title
	province_filter_target = c_jerusalem

	is_location_valid = {
		scope:province = {
			custom_tooltip = {
				text = jip_gp_temple_mount_location_tt
				this = title:c_jerusalem.title_province
			}
			custom_tooltip = {
				text = jip_gp_temple_mount_held_tt
				barony.holder ?= { target_is_same_character_or_above = root }
			}
		}
	}

	can_start_planning = {
		custom_tooltip = {
			text = jip_gp_temple_mount_jerusalem_tt
			title:c_jerusalem.holder ?= { target_is_same_character_or_above = root }
		}
		piety_level >= 4
		capital_county.rite.faith = {
			religion = religion:judaism_religion
		}
		OR = {
			has_primary_title = title:k_israel
			primary_title = {
				is_de_facto_liege_or_above_target = title:k_israel
			}
		}
		completely_controls = title:k_israel
		is_at_war = no
	}

	owner = founder_primary_title_owner

	is_valid = {
		scope:founder = {
			has_religion = religion:judaism_religion
			is_independent_ruler = yes
		}
		scope:province.barony.holder ?= { target_is_same_character_or_above = scope:founder }
	}

	cost = {
		treasury_or_gold = {
			value = great_project_base_starting_cost
		}
		piety = {
			value = 500
		}
	}

	construction_time = 1095 # 3 years

	allowed_contributor_filter = {
		owner = yes
		founder = yes
		vassals = yes
		tributaries = yes
		allies = yes
		head_of_faith = yes
	}

	contributor_cooldown = standard_contributor_cooldown_value

	on_start_build = {
		scope:founder ?= {
			jip_temple_mount_clearing_begins_effect = yes
		}
	}

	on_cancel = {
		scope:founder ?= {
			jip_temple_mount_clearing_ends_effect = yes
		}
	}

	on_complete = {
		scope:owner ?= {
			jip_temple_mount_clearing_ends_effect = yes
			jip_clear_the_temple_mount_effect = yes
			trigger_event = jip_hebrew.0041
		}
	}

	project_contributions = {
		# Required: the masons and labourers who take the dome down stone by stone
		labourers = {
			cost = {
				treasury_or_gold = {
					value = great_project_base_contribution_cost
				}
			}
			contributor_cooldown = individual_contribution_cooldown_value
			is_required = yes
			on_contribution_funded = {
				add_prestige = medium_prestige_gain
				great_project_notify_contribution = yes
			}
			on_complete = {
				great_project_reward_contribution_completion_notification_effect = {
					LEVEL = minor
					NON_OWNER_REWARD_EFFECT = great_project_reward_contribution_completion_null
					OWNER_REWARD_EFFECT = great_project_reward_contribution_completion_null
				}
			}
			ai_check_interval_by_tier = {
				barony = 0
				county = 12
				duchy = 6
				kingdom = 3
				empire = 3
				hegemony = 3
			}
			ai_will_do = { value = 50 }
		}
		# Required: the ashes of the red heifer, without which no priest may set foot on the Mount (Numbers 19)
		red_heifer = {
			contributor_is_valid = {
				custom_tooltip = {
					text = jip_gp_temple_jewish_contributor_tt
					has_religion = religion:judaism_religion
				}
			}
			cost = {
				piety = {
					value = 300
				}
			}
			contributor_cooldown = individual_contribution_cooldown_value
			is_required = yes
			on_contribution_funded = {
				add_piety = minor_piety_gain
				great_project_notify_contribution = yes
			}
			on_complete = {
				great_project_reward_contribution_completion_notification_effect = {
					LEVEL = major
					NON_OWNER_REWARD_EFFECT = great_project_reward_contribution_completion_null
					OWNER_REWARD_EFFECT = great_project_reward_contribution_completion_null
				}
			}
			ai_check_interval_by_tier = {
				barony = 0
				county = 12
				duchy = 6
				kingdom = 3
				empire = 3
				hegemony = 3
			}
			ai_will_do = { value = 50 }
		}
		# Optional: the survey of the Mount, to find where the Holy of Holies stood
		survey = {
			cost = {
				treasury_or_gold = {
					value = great_project_base_contribution_cost
					multiply = 0.5
				}
			}
			contributor_cooldown = individual_contribution_cooldown_value
			is_required = no
			on_contribution_funded = {
				add_prestige = minor_prestige_gain
				great_project_notify_contribution = yes
			}
			on_complete = {
				great_project_reward_contribution_completion_notification_effect = {
					LEVEL = minor
					NON_OWNER_REWARD_EFFECT = great_project_reward_contribution_completion_null
					OWNER_REWARD_EFFECT = great_project_reward_contribution_completion_null
				}
			}
			ai_check_interval_by_tier = {
				barony = 0
				county = 12
				duchy = 12
				kingdom = 6
				empire = 6
				hegemony = 6
			}
			ai_will_do = { value = 25 }
		}
	}

	ai_target_quick_trigger = {
		adult = yes
		rank = kingdom
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	# Rarely, and only with the zeal for it
	ai_will_do = {
		value = 0
		if = {
			limit = { has_trait = zealous }
			add = 50
		}
		if = {
			limit = { ai_zeal >= 50 }
			add = 25
		}
	}

	group = major_project
	is_important = yes
	target_title_tier = barony
}
