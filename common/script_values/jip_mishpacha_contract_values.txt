# Mishpacha contracts (common/task_contracts/jip_mishpacha_contracts.txt)

# Gold paid on success, scaled by the employer's tier like vanilla contracts (counts x1, dukes x3, kings and emperors x4, courtiers x0.5)
# scope:employer = the employer
jip_contract_gold_value = {
	value = 50
	multiply = scope:employer.task_contract_employer_tier_gold_multiplier_value
}

# Gold the family puts up front for a loan or a caravan, and pays as a ransom
# scope:employer = the employer
jip_contract_stake_value = {
	value = 60
	multiply = scope:employer.task_contract_employer_tier_gold_multiplier_value
}

# Contracts added by each refresh: three, and one more brought in by a Shaliach
# root = the family head
jip_contract_refresh_amount_value = {
	value = 3
	if = {
		limit = { jip_employs_family_envoy_trigger = yes }
		add = 1
	}
}

# Chance in percent that an approach succeeds, before the approach's own bonus (jip_contract_resolve_effect).
# The matching estate building and court position help. root = the contract taker
jip_contract_stewardship_chance_value = {
	value = 20
	add = {
		value = stewardship
		multiply = 3
	}
	if = {
		limit = { domicile ?= { has_domicile_parameter = jip_estate_counting_house_3 } }
		add = 15
	}
	else_if = {
		limit = { domicile ?= { has_domicile_parameter = jip_estate_counting_house_2 } }
		add = 10
	}
	else_if = {
		limit = { domicile ?= { has_domicile_parameter = jip_estate_counting_house_1 } }
		add = 5
	}
	if = {
		limit = { jip_employs_family_agent_trigger = yes }
		add = 10
	}
}
