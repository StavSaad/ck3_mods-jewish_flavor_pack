# The Return of the Moon God, the Sabian story (events/jip_sabian_story_events.txt)

# Chance (%) that a learned answer carries the day: 20, plus 3 per point of learning, between 10 and 90
jip_sabian_story_learning_chance_value = {
	value = 20
	add = {
		value = learning
		multiply = 3
	}
	min = 10
	max = 90
}

jip_sabian_story_learning_failure_value = {
	value = 100
	subtract = jip_sabian_story_learning_chance_value
}

### Act II: the Return to Harran

# The price of Harran's tax farm: 250 gold, plus 30 per level of development
jip_sabian_harran_price_value = {
	value = 250
	add = {
		value = title:c_harran.development_level
		multiply = 30
	}
}

# The Harranians who rise: 900 men, plus 25 per level of development
jip_sabian_harranian_levies_value = {
	value = 900
	add = {
		value = title:c_harran.development_level
		multiply = 25
	}
}

### Act III: Throw Off the Caliph

# The Harranians who fight for Harran's freedom (root = the story owner): the rising's army, plus 150 men for every other
# Sabian or old-faith county in the realm
jip_sabian_freedom_levies_value = {
	value = jip_sabian_harranian_levies_value
	every_sub_realm_county = {
		limit = {
			this != title:c_harran
			OR = {
				faith = faith:mandaean_faith
				faith = faith:mesopotamian_pagan
			}
		}
		add = 150
	}
}

### Act V: Reclaim the Land

# Chance (%) that a search finds the god's statue: 30, plus 4 per point of the chosen skill (learning or intrigue),
# between 20 and 90
jip_sabian_statue_search_chance_value = {
	value = 30
	if = {
		limit = { scope:jip_search_skill = flag:intrigue }
		add = {
			value = intrigue
			multiply = 4
		}
	}
	else = {
		add = {
			value = learning
			multiply = 4
		}
	}
	min = 20
	max = 90
}
