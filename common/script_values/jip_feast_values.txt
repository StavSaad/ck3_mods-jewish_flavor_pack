# The Mandaean calendar has 365 days and no leap day, so its feasts fall a day earlier every four years. Counted back
# from 2024 (Dehwa Rabba 16 July, Parwanaya 13 March, Gregorian), in 867 Dehwa Rabba falls on 19 April and Parwanaya on
# 15 December (Julian); by 1453 they have moved to late November and late July. CK3's calendar has no leap days, so the
# day of the year is computed from the year: the 867 day, less a day for every four years since, wrapped into 1-365.
# The feast variants (tools/make_feast_override.py) are shown in the month that day falls in

jip_dehwa_rabba_raw_day_value = {
	value = 109 # 19 April 867
	subtract = {
		value = current_year
		subtract = 867
		divide = 4
		floor = yes
	}
}

jip_dehwa_rabba_day_value = {
	value = jip_dehwa_rabba_raw_day_value
	if = {
		limit = { jip_dehwa_rabba_raw_day_value < 1 }
		add = 365
	}
	else_if = {
		limit = { jip_dehwa_rabba_raw_day_value > 365 }
		subtract = 365
	}
}

jip_dehwa_rabba_month_value = {
	value = 1
	if = {
		limit = { jip_dehwa_rabba_day_value >= 32 }
		value = 2
	}
	if = {
		limit = { jip_dehwa_rabba_day_value >= 60 }
		value = 3
	}
	if = {
		limit = { jip_dehwa_rabba_day_value >= 91 }
		value = 4
	}
	if = {
		limit = { jip_dehwa_rabba_day_value >= 121 }
		value = 5
	}
	if = {
		limit = { jip_dehwa_rabba_day_value >= 152 }
		value = 6
	}
	if = {
		limit = { jip_dehwa_rabba_day_value >= 182 }
		value = 7
	}
	if = {
		limit = { jip_dehwa_rabba_day_value >= 213 }
		value = 8
	}
	if = {
		limit = { jip_dehwa_rabba_day_value >= 244 }
		value = 9
	}
	if = {
		limit = { jip_dehwa_rabba_day_value >= 274 }
		value = 10
	}
	if = {
		limit = { jip_dehwa_rabba_day_value >= 305 }
		value = 11
	}
	if = {
		limit = { jip_dehwa_rabba_day_value >= 335 }
		value = 12
	}
}

# Years since the last Akitu (jip_akitu_last_year, set when one is held, or when a ruler first comes under the rule)

	value = current_year
	subtract = var:jip_akitu_last_year
}

# The household feast (jip_household_feast_schedule_effect) is sent from the yearly pulse to land in its season: the
# seder (jip_feast.5020) between days 100 and 120 of the year (10-30 April; Pesach, 15 Nisan, falls from late March to
# late April), Purim (7020) between 60 and 80 (1-21 March; 14 Adar falls from late February to late March), Sukkot
# (8020) between 265 and 285 (22 September to 12 October; 15 Tishri falls from mid-September to mid-October), Hanukkah
# (9020) between 335 and 355 (1-21 December; 25 Kislev falls from late November to late December)
jip_day_of_year_value = {
	value = current_day
	if = {
		limit = { current_month = 2 }
		add = 31
	}
	else_if = {
		limit = { current_month = 3 }
		add = 59
	}
	else_if = {
		limit = { current_month = 4 }
		add = 90
	}
	else_if = {
		limit = { current_month = 5 }
		add = 120
	}
	else_if = {
		limit = { current_month = 6 }
		add = 151
	}
	else_if = {
		limit = { current_month = 7 }
		add = 181
	}
	else_if = {
		limit = { current_month = 8 }
		add = 212
	}
	else_if = {
		limit = { current_month = 9 }
		add = 243
	}
	else_if = {
		limit = { current_month = 10 }
		add = 273
	}
	else_if = {
		limit = { current_month = 11 }
		add = 304
	}
	else_if = {
		limit = { current_month = 12 }
		add = 334
	}
}

jip_days_to_pesach_value = {
	value = 100
	subtract = jip_day_of_year_value
	if = {
		limit = { jip_day_of_year_value >= 100 }
		add = 365
	}
}

# The household feast's rotation (jip_household_feast_schedule_effect): years since the character last kept each feast,
# at a feast or at home (var:jip_last_kept_<feast>, jip_feast_mark_kept_effect), 1000 if never, -1 if the faith doesn't
# keep it; the fraction breaks ties (Pesach, Sukkot, Hanukkah, Purim)
jip_household_pesach_wait_value = {
	value = 1000
	if = {
		limit = { has_variable = jip_last_kept_pesach }
		value = current_year
		subtract = var:jip_last_kept_pesach
	}
	add = 0.4
}
