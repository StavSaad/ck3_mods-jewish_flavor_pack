# Kehunah: the hereditary priesthood of the sons of Aaron, who serve in the Temple and live from the tithes of Israel.
tenet_kehunah = {
	icon = tenet_kehunah # The High Priest's breastplate
	piety_cost = {
		value = faith_tenet_cost_mid
		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_kehunah }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}
	# Only for Jewish faiths, and faiths that take in Jewish ways through Jewish Syncretism
	can_pick = {
		custom_description = {
			text = jip_tenet_kehunah_jewish_trigger
			OR = {
				religion = religion:judaism_religion
				tenet:tenet_jewish_syncretism = { is_in_list = selected_doctrines }
			}
		}
	}
	can_pick_as_personal_tenet = {
		OR = {
			has_religion = religion:judaism_religion
			rite = { rite_has_tenet = tenet_jewish_syncretism }
		}
	}
	character_modifier = {
		monthly_piety_gain_mult = 0.1
		clergy_opinion = 10
	}
	# The tithes owed to the priests and Levites
	county_modifier = {
		tax_mult = 0.05
	}
	traits = {
		virtues = {
			chaste = { weight = 15 scale = 1 } # Laws of priestly purity
			zealous = { weight = 15 scale = 1 }
		}
		sins = {
			lustful = { weight = 15 scale = 1 }
			cynical = { weight = 15 scale = 1 }
		}
	}
}
