# Sinism's children: the Aramean cities and the star-readers take the crescent standard of Sin
jip_harranian_rite = {
	trigger = {
		exists = scope:founder
		scope:faith = faith:mesopotamian_pagan
		OR = {
			scope:founder.culture ?= culture:syriac
			rite_has_tenet = tenet_astrology
		}
	}

	custom_faith_icons = { harranian_01 harranian_02 harranian_03 harranian_04 harranian_05 harranian_06 }
}
