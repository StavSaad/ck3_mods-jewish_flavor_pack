# The Sabian story's current act, for its story window (JIP_SABIAN_STORY_ACT); keyed on the story owner
JipSabianStoryAct = {
	type = character
	text = {
		trigger = {
			any_owned_story = {
				type = story_jip_sabian_revival
				var:jip_sabian_act ?= 6
			}
		}
		localization_key = jip_sabian_story_act_6
	}
	text = {
		trigger = {
			any_owned_story = {
				type = story_jip_sabian_revival
				var:jip_sabian_act ?= 5
			}
		}
		localization_key = jip_sabian_story_act_5
	}
	text = {
		trigger = {
			any_owned_story = {
				type = story_jip_sabian_revival
				var:jip_sabian_act ?= 4
				has_variable = jip_sabian_unveiled
			}
		}
		localization_key = jip_sabian_story_act_4_unveiled
	}
	text = {
		trigger = {
			any_owned_story = {
				type = story_jip_sabian_revival
				var:jip_sabian_act ?= 4
			}
		}
		localization_key = jip_sabian_story_act_4
	}
	text = {
		trigger = {
			any_owned_story = {
				type = story_jip_sabian_revival
				var:jip_sabian_act ?= 3
			}
		}
		localization_key = jip_sabian_story_act_3
	}
	text = {
		trigger = {
			any_owned_story = {
				type = story_jip_sabian_revival
				var:jip_sabian_act ?= 2
			}
		}
		localization_key = jip_sabian_story_act_2
	}
	text = {
		trigger = { always = no }
		localization_key = jip_sabian_story_act_1
		fallback = yes
	}
}

# The six gods' cities in the story window in Acts IV and V (JIP_SABIAN_STORY_ACT), keyed on the story owner: the
# statue home, the temple standing in the realm, the temple standing outside it, the city in the realm without its
# temple, or the city outside the realm. Generated with the header JipSabianSitesHeader; empty in the other acts
JipSabianSitesHeader = {
	type = character
	text = {
		trigger = {
			any_owned_story = {
				type = story_jip_sabian_revival
				OR = {
					var:jip_sabian_act ?= 4
					var:jip_sabian_act ?= 5
				}
			}
		}
		localization_key = jip_sabian_sites_header
	}
	text = {
		localization_key = jip_sabian_empty
	}
}

JipSabianSiteSin = {
	type = character
	text = {
		trigger = {
			NOT = {
				any_owned_story = {
					type = story_jip_sabian_revival
					OR = {
						var:jip_sabian_act ?= 4
						var:jip_sabian_act ?= 5
					}
				}
			}
		}
		localization_key = jip_sabian_empty
	}
	text = {
		trigger = {
			is_target_in_global_variable_list = { name = jip_old_gods_home target = flag:sin }
		}
		localization_key = jip_sabian_site_sin_home
	}
	text = {
		trigger = {
			title:b_harran.title_province = { has_building = jip_temple_of_sin_01 }
			title:b_harran.holder ?= { target_is_same_character_or_above = root }
		}
		localization_key = jip_sabian_site_sin_temple
	}
	text = {
		trigger = {
			title:b_harran.title_province = { has_building = jip_temple_of_sin_01 }
		}
		localization_key = jip_sabian_site_sin_temple_lost
	}
	text = {
		trigger = {
			title:b_harran.holder ?= { target_is_same_character_or_above = root }
		}
		localization_key = jip_sabian_site_sin_held
	}
	text = {
		localization_key = jip_sabian_site_sin_lost
	}
}
