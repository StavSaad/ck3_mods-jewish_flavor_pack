# Mishpacha contracts (common/task_contracts/jip_mishpacha_contracts.txt)

# Character scope: a family head who is offered Mishpacha contracts. Players only, like vanilla's adventurer contracts;
# AI families get jip_mishpacha_ai_contract_effect instead. Jewish families only: the contracts' texts are Jewish
# (the Sabian and Mandaean families' contracts are still to come, 0.15 plan section 8)
jip_mishpacha_contract_owner_trigger = {
	jip_is_jewish_family_trigger = yes
	is_ai = no
	has_ep3_dlc_trigger = yes
	exists = domicile
}

# Character scope: can offer or keep offering a Mishpacha contract. Players never give contracts, like vanilla
jip_mishpacha_contract_employer_trigger = {
	is_ai = yes
	is_alive = yes
	is_adult = yes
	is_incapable = no
	is_imprisoned = no
	NOR = {
		government_has_flag = government_is_landless_adventurer
		government_has_flag = government_is_herder
	}
}

# Contract creation: root = the family head, scope:employer = the employer
jip_mishpacha_contract_create_trigger = {
	jip_mishpacha_contract_owner_trigger = yes
	scope:employer = {
		jip_mishpacha_contract_employer_trigger = yes
		NOT = { this = root }
		NOT = { has_relation_rival = root }
	}
	# One offer per employer at a time
	NOT = {
		any_character_task_contract = { task_contract_employer ?= scope:employer }
	}
}

# Accepting: root = the family head, scope:employer = the employer. One contract at a time, two while a Sofer keeps the books
jip_mishpacha_contract_accept_trigger = {
	jip_mishpacha_contract_owner_trigger = yes
	is_imprisoned = no
	custom_tooltip = {
		text = jip_mishpacha_contract_busy_tt
		OR = {
			num_taken_task_contracts < 1
			AND = {
				num_taken_task_contracts < 2
				employs_court_position = jip_sofer_court_position
			}
		}
	}
	scope:employer = { jip_mishpacha_contract_employer_trigger = yes }
}

# An offer not yet taken stays on the board for about a year: root = the contract
jip_mishpacha_contract_keep_trigger = {
	has_variable = jip_contract_fresh
	task_contract_employer ?= { jip_mishpacha_contract_employer_trigger = yes }
}

# A taken contract carries on: root = the contract
jip_mishpacha_contract_continue_trigger = {
	task_contract_taker ?= {
		is_alive = yes
		government_has_flag = government_is_familial
	}
	task_contract_employer ?= { is_alive = yes }
}

# A contract event can fire: root = the family head, scope:task_contract = the contract
jip_mishpacha_contract_event_trigger = {
	exists = scope:task_contract
	scope:task_contract = { task_contract_taker ?= root }
}

# The contract's rolled outcome is $OUTCOME$ (critical, success or failure): scope:task_contract = the contract
jip_contract_outcome_is_trigger = {
	scope:task_contract.var:jip_contract_outcome ?= flag:$OUTCOME$
}

# The rolled outcome is a success of either kind
jip_contract_succeeded_trigger = {
	OR = {
		jip_contract_outcome_is_trigger = { OUTCOME = critical }
		jip_contract_outcome_is_trigger = { OUTCOME = success }
	}
}

# The family head chose the $APPROACH$ approach
jip_contract_approach_is_trigger = {
	scope:task_contract.var:jip_contract_approach ?= flag:$APPROACH$
}

# A contract scheme can go on: scope:owner = the family head, scope:target = its target
jip_contract_scheme_valid_trigger = {
	scope:owner = {
		government_has_flag = government_is_familial
		is_imprisoned = no
	}
	scope:target = { is_alive = yes }
}

# The character a contract is about (the envoy's counterpart, a captive, a patient, a rival) is still there: root = the contract
jip_mishpacha_contract_target_alive_trigger = {
	var:jip_contract_target ?= { is_alive = yes }
}

### Employers of the individual contracts (character scope = the employer)

# Extend a Loan: short of money, or paying for a war
jip_contract_loan_employer_trigger = {
	is_landed = yes
	highest_held_title_tier >= tier_county
	OR = {
		treasury_or_gold < 100
		is_at_war = yes
	}
}

# Farm the Tolls: holds a city in the domain
jip_contract_tolls_employer_trigger = {
	is_landed = yes
	highest_held_title_tier >= tier_county
	any_directly_owned_province = { has_holding_type = city_holding }
}

# Finance a Caravan: a ruler with money to put into a venture
jip_contract_caravan_employer_trigger = {
	is_landed = yes
	highest_held_title_tier >= tier_county
	treasury_or_gold >= 100
}

# Act as Envoy: an independent ruler bordering a realm of another faith
jip_contract_envoy_employer_trigger = {
	is_landed = yes
	is_independent_ruler = yes
	highest_held_title_tier >= tier_duchy
	any_neighboring_top_liege_realm_owner = { jip_contract_envoy_counterpart_trigger = { EMPLOYER = prev } }
}

jip_contract_envoy_counterpart_trigger = {
	is_ai = yes
	NOT = { religion = $EMPLOYER$.religion }
	NOT = { is_at_war_with = $EMPLOYER$ }
}

# Redeem the Captives: holds a Jew who is not a ruler in the dungeon
jip_contract_ransom_employer_trigger = {
	is_landed = yes
	NOT = { has_religion = religion:judaism_religion }
	any_prisoner = { jip_ransomable_captive_trigger = yes }
}

# The captive is still alive and still in the employer's dungeon: root = the contract
jip_contract_ransom_captive_held_trigger = {
	var:jip_contract_target ?= {
		is_alive = yes
		is_imprisoned = yes
		imprisoner ?= root.task_contract_employer
	}
}

# Serve as Court Physician: the ruler or close family is ill and has no good treatment yet
jip_contract_physician_employer_trigger = {
	is_landed = yes
	OR = {
		jip_contract_physician_patient_trigger = yes
		any_close_family_member = { jip_contract_physician_patient_trigger = yes }
	}
}

jip_contract_physician_patient_trigger = {
	is_alive = yes
	has_treatable_disease_trigger = yes
	NOT = { has_good_disease_treatment_trigger = yes }
}

# Translate Texts: a learned ruler of another culture
jip_contract_translate_employer_trigger = {
	is_landed = yes
	highest_held_title_tier >= tier_county
	NOT = { culture = $OWNER$.culture }
	OR = {
		learning >= 12
		has_trait = erudite
		has_trait = theologian
		has_trait = lifestyle_physician
	}
}

# Answer a Responsum: a Jew of another house who needs a ruling
jip_contract_responsum_employer_trigger = {
	has_religion = religion:judaism_religion
	NOT = { house ?= $OWNER$.house }
}

# Gather News from Correspondents: a ruler who wants to know what a rival is up to
jip_contract_news_employer_trigger = {
	is_landed = yes
	highest_held_title_tier >= tier_county
	any_relation = {
		type = rival
		jip_contract_news_target_trigger = { OWNER = $OWNER$ }
	}
}

jip_contract_news_target_trigger = {
	is_alive = yes
	NOT = { this = $OWNER$ }
}
