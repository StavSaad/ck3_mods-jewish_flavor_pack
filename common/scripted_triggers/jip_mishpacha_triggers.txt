# Title scope: a Jewish family title: one of the historical d_jip_* titles, or a noble family title given to a dispossessed lord
jip_is_mishpacha_family_title_trigger = {
	OR = {
		is_noble_family_title = yes
		this = title:d_jip_ananite
		this = title:d_jip_benveniste
		this = title:d_jip_bostanite
		this = title:d_jip_kalonymos
		this = title:d_jip_maimon
		this = title:d_jip_makhirite
		this = title:d_jip_nathanite
		this = title:d_jip_shealtiel
	}
}

# Character scope: an AI Mishpacha family that is free to consider moving its estate to another realm
jip_mishpacha_ai_can_migrate_trigger = {
	is_ai = yes
	has_government = mishpacha_government
	is_landed = no
	is_adult = yes
	is_available = yes
	is_at_war = no
	NOR = {
		# Expelled families have their own journey
		has_variable = jip_expelled_by
		has_variable = jip_expulsion_refuge
		has_variable = jip_mishpacha_migration_cooldown
		# The Exilarchs keep their seat in Babylon
		has_title = title:d_exilarchate
		has_title = title:d_karaite_exilarchate
	}
	exists = domicile
	location = domicile.domicile_location
	# AI families wait out a war before moving, though a player family may leave during one (jip_scripted_rules.txt)
	domicile.domicile_location.county.holder ?= {
		NOR = {
			is_at_war = yes
			any_liege_or_above = { is_at_war = yes }
		}
	}
}

# Character scope: may build the Mishpacha branches of the estate chains (jip_mishpacha_estate_buildings.txt)
jip_can_build_mishpacha_estate_building_trigger = {
	government_has_flag = government_is_mishpacha
}
