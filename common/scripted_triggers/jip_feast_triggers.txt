# The mod's festivals: feast variants added by tools/make_feast_override.py (0.15 plan, sections 7 and 9)

# Who may host a feast: vanilla's rule, plus landless Mishpacha heads, who hold it at their estate
jip_can_host_feast_trigger = {
	OR = {
		is_landed_or_landless_administrative_or_theocratic = yes
		government_has_flag = government_is_familial
	}
}

# Which variant scope:activity is, like vanilla's is_tsagaan_sar
jip_is_akitu = {
	scope:activity = {
		has_activity_option = {
			category = special_type
			option = jip_feast_type_akitu
		}
	}
}
jip_is_nisan = {
	scope:activity = {
		has_activity_option = {
			category = special_type
			option = jip_feast_type_nisan
		}
	}
}
jip_is_parwanaya = {
	scope:activity = {
		has_activity_option = {
			category = special_type
			option = jip_feast_type_parwanaya
		}
	}
}
jip_is_dehwa_rabba = {
	scope:activity = {
		has_activity_option = {
			category = special_type
			option = jip_feast_type_dehwa_rabba
		}
	}
}

jip_is_pesach = {
	scope:activity = {
		has_activity_option = {
			category = special_type
			option = jip_feast_type_pesach
		}
	}
}

jip_is_festival = {
	OR = {
		jip_is_akitu = yes
		jip_is_nisan = yes
		jip_is_parwanaya = yes
		jip_is_dehwa_rabba = yes
		jip_is_pesach = yes
	}
}

# The festival is held at a holy site of the host's faith (Harran, Nineveh, Arbela for Akitu)
jip_feast_at_holy_site_trigger = {
	scope:activity.activity_location = {
		barony = { is_holy_site_of = scope:host.faith }
	}
}

# A Mesopotamian ruler expected to keep Akitu: a duke or above
jip_akitu_keeper_trigger = {
	is_ruler = yes
	is_landed = yes
	highest_held_title_tier >= tier_duchy
	faith = faith:mesopotamian_pagan
}

# Pesach (0.16 plan, section 1). Character scope: keeps the seder of the rabbinic Haggadah (the four questions, the
# afikoman, Elijah's cup). The Karaites read the Exodus from the Torah alone, and the Samaritans and the Beta Israel
# still sacrifice the paschal lamb
jip_pesach_haggadah_trigger = {
	religion = religion:judaism_religion
	NOR = {
		faith = faith:karaism
		faith = faith:samaritan
		faith = faith:haymanot
	}
}

# Character scope: still sacrifices the paschal lamb (the Samaritans on Mount Gerizim, the Beta Israel)
jip_pesach_paschal_lamb_trigger = {
	OR = {
		faith = faith:samaritan
		faith = faith:haymanot
	}
}

# The Temple in Jerusalem stands again (temple_in_jerusalem_01, rebuild_the_third_temple_effect): the Korban Pesach
jip_temple_stands_trigger = {
	title:c_jerusalem.title_province = { has_building = temple_in_jerusalem_01 }
}

# scope:host holds Jerusalem, directly or through a vassal: "this year in Jerusalem"
jip_pesach_host_holds_jerusalem_trigger = {
	title:c_jerusalem.holder ?= { target_is_same_character_or_above = scope:host }
}

# Character scope: an Exilarch of either line
jip_is_exilarch_trigger = {
	OR = {
		has_title = title:d_exilarchate
		has_title = title:d_karaite_exilarchate
	}
}
