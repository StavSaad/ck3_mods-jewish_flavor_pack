# Who may host a feast: vanilla's rule, plus landless Mishpacha heads, who hold it at their estate
jip_can_host_feast_trigger = {
	OR = {
		can_host_feasts_trigger = yes # vanilla's rule: landed, landless administrative, merchants, republics
		government_has_flag = government_is_familial
	}
}

# The festival is held at a holy site of the host's faith (Harran, Nineveh, Arbela for Akitu)
jip_feast_at_holy_site_trigger = {
	scope:activity.activity_location = {
		barony = { is_holy_site_of = scope:host.faith }
	}
}

# A Mesopotamian ruler expected to keep Akitu: a duke or above
jip_akitu_keeper_trigger = {
	is_ruler = yes
	is_landed = yes
	highest_held_title_tier >= tier_duchy
	faith = faith:mesopotamian_pagan
}

# Pesach (0.16 plan, section 1). Character scope: keeps the seder of the rabbinic Haggadah (the four questions, the
# afikoman, Elijah's cup). The Karaites read the Exodus from the Torah alone, and the Samaritans and the Beta Israel
# still sacrifice the paschal lamb
jip_pesach_haggadah_trigger = {
	religion = religion:judaism_religion
	NOR = {
		faith = faith:karaism
		faith = faith:samaritan
		faith = faith:haymanot
	}
}

# Character scope: still sacrifices the paschal lamb (the Samaritans on Mount Gerizim, the Beta Israel)
jip_pesach_paschal_lamb_trigger = {
	OR = {
		faith = faith:samaritan
		faith = faith:haymanot
	}
}

# The Temple in Jerusalem stands again (jip_temple_in_jerusalem_01, jip_gp_rebuild_the_temple): the Korban Pesach
jip_temple_stands_trigger = {
	title:c_jerusalem.title_province = { has_building = jip_temple_in_jerusalem_01 }
}

# scope:host holds Jerusalem, directly or through a vassal: "this year in Jerusalem"
jip_pesach_host_holds_jerusalem_trigger = {
	title:c_jerusalem.holder ?= { target_is_same_character_or_above = scope:host }
}

# Character scope: an Exilarch of either line
jip_is_exilarch_trigger = {
	OR = {
		has_title = title:d_exilarchate
		has_title = title:d_karaite_exilarchate
	}
}

# Purim, Sukkot and Hanukkah (0.16 plan, section 2)

# Character scope: keeps the rabbinic feasts, Hanukkah among them: every Jewish faith but the Karaites, the Samaritans
# and the Beta Israel, who keep the Torah's feasts alone and never took up the sages' Hanukkah. The same faiths read the
# rabbinic Haggadah at Pesach
jip_keeps_rabbinic_feasts_trigger = {
	jip_pesach_haggadah_trigger = yes
}

# Character scope: keeps Purim: the rabbinic faiths and the Karaites, who keep it for the Scroll of Esther, though with
# their own calendar and none of the sages' rules (the Samaritans' Bible has no Esther, and the Beta Israel never kept it)
jip_keeps_purim_trigger = {
	OR = {
		jip_keeps_rabbinic_feasts_trigger = yes
		faith = faith:karaism
	}
}

# Character scope: the character's liege or suzerain is a gentile who bears the Jews ill will: hostile by nature
# (jip_jewish_hostility_value, the Expel the Jews weights), or hostile to this character, or ruling a realm that
# expelled the Jews. Purim's Haman (jip_feast.7001), Hanukkah's lights kept within (jip_feast.9001)
jip_hostile_gentile_overlord_trigger = {
	save_temporary_scope_as = jip_overlord_subject
	OR = {
		liege ?= {
			NOT = { this = scope:jip_overlord_subject }
			jip_is_hostile_gentile_overlord_trigger = { SUBJECT = scope:jip_overlord_subject }
		}
		suzerain ?= {
			jip_is_hostile_gentile_overlord_trigger = { SUBJECT = scope:jip_overlord_subject }
		}
	}
}

# Character scope (the overlord): a gentile hostile to the Jews or to $SUBJECT$
jip_is_hostile_gentile_overlord_trigger = {
	NOT = { religion = religion:judaism_religion }
	OR = {
		jip_jewish_hostility_value >= 30
		opinion = {
			target = $SUBJECT$
			value <= -25
		}
		jip_ruler_has_expelled_jews_trigger = yes
	}
}

# The feast (scope:activity) is held in the city of Hamadan, where the Jews of Persia keep the tomb of Esther and Mordecai
jip_feast_at_hamadan_trigger = {
	scope:activity.activity_location ?= { geographical_region = jip_city_hamadan }
}

# Character scope: is in a land where the citron grows (the Levant, Egypt, the Maghreb, Iberia, Italy and
# the Greek lands), so the etrog for Sukkot needs no buying (jip_feast.8002)
jip_sukkot_citron_land_trigger = {
	location ?= { geographical_region = jip_region_citron_lands }
}

# Character scope: an etrog comes without buying: a Shaliach of the court brings one from the Land, or a merchant
# company's ships carry them
jip_sukkot_etrog_provided_trigger = {
	OR = {
		employs_court_position = jip_shaliach_court_position
		government_has_flag = government_is_merchant
	}
}

# Character scope: keeps Sukkot in the Land of Israel before about 1200, when its synagogues still read the Torah over
# three years and had no Simchat Torah (jip_feast.8004)
jip_sukkot_palestinian_rite_trigger = {
	current_year < 1200
	location ?= { geographical_region = jip_region_israel }
}

# Character scope: of a house descended from the Hasmoneans (0.16 plan, section 3; the house sets the variable when it
# is built)
jip_is_hasmonean_heir_trigger = {
	house ?= { has_variable = jip_hasmonean_house }
}

# Purim: the feast (scope:activity) pours wine, normal or extensive, for ad de-lo yada (jip_feast.7003)
jip_purim_wine_trigger = {
	scope:activity = {
		OR = {
			has_activity_option = {
				category = feast_option_wine
				option = feast_wine_normal
			}
			has_activity_option = {
				category = feast_option_wine
				option = feast_wine_extensive
			}
		}
	}
}

# Purim: a guest at the feast (scope:activity) keeps a secret that root does not know (jip_feast.7003)
jip_purim_guest_secret_trigger = {
	scope:activity = {
		any_attending_character = {
			NOT = { this = root }
			any_secret = { NOT = { is_known_by = root } }
		}
	}
}
