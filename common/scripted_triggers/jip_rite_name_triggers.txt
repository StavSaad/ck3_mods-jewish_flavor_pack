# Which god a new rite of Canaanism, Mesopotamianism or Arameanism is named after (common/religion/rite_names/00_jip_rite_names.txt),
# like vanilla's Tyrism and Odinite for Asatru. Root is the rite being created, scope:faith its faith, scope:founder its
# founder (may not exist). A name goes to the first entry whose trigger is met and whose name is still unused, so the
# names file lists the cult cities first, then the tenets. Canaanism has no catch-all god: it is Baal's faith already

jip_rite_founder_capital_trigger = {
	scope:founder ?= {
		capital_county ?= title:$COUNTY$
	}
}

jip_rite_founder_capital_under_trigger = {
	scope:founder ?= {
		capital_county ?= { target_is_de_jure_liege_or_above = title:$TITLE$ }
	}
}

# Marduk of Esagila, lord of Babylon
jip_rite_marduk_trigger = {
	scope:faith = faith:mesopotamian_pagan
	jip_rite_founder_capital_under_trigger = { TITLE = k_mesopotamia }
}

# Shamash of Hatra, the sun and the judge
jip_rite_shamash_trigger = {
	scope:faith = faith:mesopotamian_pagan
	OR = {
		jip_rite_founder_capital_trigger = { COUNTY = c_mosul }
		rite_has_tenet = tenet_sun_worship
		rite_has_tenet = tenet_legalism
	}
}

# A founder ruling from one of the Aramean gods' cities: the tenets' names below wait for founders elsewhere, so a
# city's own god comes first
jip_rite_aramean_cult_city_trigger = {
	OR = {
		jip_rite_founder_capital_under_trigger = { TITLE = d_damascus }
		jip_rite_founder_capital_trigger = { COUNTY = c_palmyra }
		jip_rite_founder_capital_trigger = { COUNTY = c_aleppo }
		jip_rite_founder_capital_trigger = { COUNTY = c_homs }
		jip_rite_founder_capital_trigger = { COUNTY = c_rahba }
		jip_rite_founder_capital_under_trigger = { TITLE = d_shammar }
		jip_rite_founder_capital_under_trigger = { TITLE = d_eastern_desert }
		scope:founder ?= { has_title = title:e_palmyra }
	}
}

# Hadad-Rimmon of Damascus (Rimmonism: Hadadism is the faith's main rite), or a war god's rite elsewhere
jip_rite_rimmon_trigger = {
	scope:faith = faith:aramean_pagan
	OR = {
		jip_rite_founder_capital_under_trigger = { TITLE = d_damascus }
		AND = {
			jip_rite_aramean_cult_city_trigger = no
			has_wargod_tenet_trigger = yes
		}
	}
}
