# The Return of the Moon God, the Sabian story (common/story_cycles/jip_story_sabian_revival.txt)

# Character scope: may carry the story: a Sabian, or a believer in the old gods once they are restored
jip_sabian_story_keeper_trigger = {
	OR = {
		rite = rite:sabianism
		faith = faith:mesopotamian_pagan
	}
}

# Character scope: owns the Sabian story
jip_owns_sabian_story_trigger = {
	any_owned_story = { type = story_jip_sabian_revival }
}

# Character scope: the story owner's liege or suzerain is a Muslim (the court where the Sabians must win their standing)
jip_sabian_story_has_muslim_patron_trigger = {
	any_owned_story = {
		type = story_jip_sabian_revival
		var:jip_sabian_patron ?= { religion = religion:islam_religion }
	}
}

# Story scope: at least two Sabian Court Astrologers serve Muslim rulers (jip_sabian_astrologer_court_position)
jip_sabian_astrologers_at_muslim_courts_trigger = {
	any_ruler = {
		count >= 2
		religion = religion:islam_religion
		any_court_position_holder = {
			type = jip_sabian_astrologer_court_position
			rite = rite:sabianism
		}
	}
}

# Character scope: can be initiated into the old gods' mysteries (Keep the Mysteries)
jip_sabian_initiate_candidate_trigger = {
	is_adult = yes
	is_imprisoned = no
	rite = rite:sabianism
	NOT = { exists = secret_faith }
	NOT = { any_secret = { type = secret_crypto_religionist } }
}

# Character scope ($HEAD$ = the story owner): the head, or an adult of the house or the head's spouse at the head's court,
# who can be initiated
jip_sabian_initiate_of_house_trigger = {
	jip_sabian_initiate_candidate_trigger = yes
	OR = {
		this = $HEAD$
		AND = {
			is_courtier_of = $HEAD$
			OR = {
				house ?= $HEAD$.house
				is_spouse_of = $HEAD$
			}
		}
	}
}

# Character scope: an initiate whose secret faith a Muslim has learnt, not yet answered for (jip_sabian_story.1040)
jip_sabian_initiate_exposed_trigger = {
	is_alive = yes
	NOT = { has_variable = jip_sabian_exposure_answered }
	any_secret = {
		type = secret_crypto_religionist
		any_secret_knower = { religion = religion:islam_religion }
	}
}

### Act II: the Return to Harran

# Character scope: owns the Sabian story in act $ACT$
jip_sabian_story_in_act_trigger = {
	custom_tooltip = {
		text = jip_sabian_story_in_act_$ACT$_tt
		any_owned_story = {
			type = story_jip_sabian_revival
			var:jip_sabian_act ?= $ACT$
		}
	}
}

# Character scope: a story owner in Act II who doesn't hold Harran yet, free to seek it
jip_sabian_can_seek_harran_trigger = {
	jip_sabian_story_in_act_trigger = { ACT = 2 }
	NOT = { has_title = title:c_harran }
	exists = title:c_harran.holder
}

# Character scope (the story owner): Harran lies in the patron's realm, so the patron can give it (Petition for Harran)
jip_sabian_patron_can_give_harran_trigger = {
	any_owned_story = {
		type = story_jip_sabian_revival
		exists = var:jip_sabian_patron
		title:c_harran.holder ?= {
			OR = {
				this = prev.var:jip_sabian_patron
				target_is_liege_or_above = prev.var:jip_sabian_patron
			}
		}
	}
}

# A crisis in Harran's realm, when the Harranians could be raised: its count or the count's top liege is at war
jip_sabian_harran_crisis_trigger = {
	title:c_harran.holder ?= {
		OR = {
			is_at_war = yes
			top_liege = { is_at_war = yes }
		}
	}
}

# Harran itself still keeps the Sabian faith
jip_sabian_harran_is_sabian_trigger = {
	title:c_harran = {
		OR = {
			faith = faith:mandaean_faith
			faith = faith:mesopotamian_pagan
		}
	}
}

### Act III: Throw Off the Caliph

# Character scope: a story owner in Act III who holds Harran under a liege, and can demand to be free of them
jip_sabian_can_proclaim_freedom_trigger = {
	jip_sabian_story_in_act_trigger = { ACT = 3 }
	has_title = title:c_harran
	is_independent_ruler = no
	exists = liege
}

### Act IV: Restore the Old Gods

# Character scope: owns the Sabian story and has unveiled the mysteries
jip_sabian_story_unveiled_trigger = {
	any_owned_story = {
		type = story_jip_sabian_revival
		has_variable = jip_sabian_unveiled
	}
}

# The Temple of Sin (jip_temple_of_sin_01) stands at Harran
jip_sabian_temple_of_sin_stands_trigger = {
	title:b_harran.title_province = { has_building = jip_temple_of_sin_01 }
}

### Act V: Reclaim the Land

# The old cult cities whose counties the story may claim: Harran, Arbela, Nineveh, Mosul (Ashur and Hatra), Palmyra and
# Kufa, where Babylon lies (b_al-hila, vanilla's Babylon holy site)
jip_sabian_is_cult_city_trigger = {
	OR = {
		this = title:c_harran
		this = title:c_irbil
		this = title:c_ninive
		this = title:c_mosul
		this = title:c_palmyra
		this = title:c_kufa
	}
}

# Title scope (a cult city's county; $OWNER$ = the story owner): outside the owner's realm and not held by the old faith
jip_sabian_cult_city_to_claim_trigger = {
	jip_sabian_is_cult_city_trigger = yes
	holder ?= {
		NOT = { target_is_same_character_or_above = $OWNER$ }
		NOT = { faith = faith:mesopotamian_pagan }
	}
	NOT = { $OWNER$ = { has_claim_on = prev } }
}

# Character scope: a cult city they could claim
jip_sabian_has_cult_city_to_claim_trigger = {
	OR = {
		title:c_harran = { jip_sabian_cult_city_to_claim_trigger = { OWNER = root } }
		title:c_irbil = { jip_sabian_cult_city_to_claim_trigger = { OWNER = root } }
		title:c_ninive = { jip_sabian_cult_city_to_claim_trigger = { OWNER = root } }
		title:c_mosul = { jip_sabian_cult_city_to_claim_trigger = { OWNER = root } }
		title:c_palmyra = { jip_sabian_cult_city_to_claim_trigger = { OWNER = root } }
		title:c_kufa = { jip_sabian_cult_city_to_claim_trigger = { OWNER = root } }
	}
}

# Character scope: the god $GOD$'s statue can be sought: not yet home, and its restored temple ($BUILDING$ at $BARONY$)
# stands in the realm
jip_sabian_god_statue_can_be_sought_trigger = {
	NOT = {
		is_target_in_global_variable_list = {
			name = jip_old_gods_home
			target = flag:$GOD$
		}
	}
	title:$BARONY$.title_province = { has_building = $BUILDING$ }
	title:$BARONY$.holder ?= { target_is_same_character_or_above = root }
}

# Character scope: a god's statue can be sought
jip_sabian_any_statue_can_be_sought_trigger = {
	OR = {
		jip_sabian_god_statue_can_be_sought_trigger = { GOD = sin BUILDING = jip_temple_of_sin_01 BARONY = b_harran }
		jip_sabian_god_statue_can_be_sought_trigger = { GOD = ishtar BUILDING = jip_temple_of_ishtar_01 BARONY = b_irbil }
		jip_sabian_god_statue_can_be_sought_trigger = { GOD = ashur BUILDING = jip_temple_of_ashur_01 BARONY = b_ain_al-qayyara }
		jip_sabian_god_statue_can_be_sought_trigger = { GOD = bel BUILDING = jip_temple_of_bel_01 BARONY = b_tadmur }
		jip_sabian_god_statue_can_be_sought_trigger = { GOD = shamash BUILDING = jip_temple_of_shamash_01 BARONY = b_al-hadr }
		jip_sabian_god_statue_can_be_sought_trigger = { GOD = marduk BUILDING = jip_temple_of_marduk_01 BARONY = b_al-hila }
	}
}

# All six gods are home: the story's ending
jip_sabian_all_gods_home_trigger = {
	is_target_in_global_variable_list = { name = jip_old_gods_home target = flag:sin }
	is_target_in_global_variable_list = { name = jip_old_gods_home target = flag:ishtar }
	is_target_in_global_variable_list = { name = jip_old_gods_home target = flag:ashur }
	is_target_in_global_variable_list = { name = jip_old_gods_home target = flag:bel }
	is_target_in_global_variable_list = { name = jip_old_gods_home target = flag:shamash }
	is_target_in_global_variable_list = { name = jip_old_gods_home target = flag:marduk }
}
