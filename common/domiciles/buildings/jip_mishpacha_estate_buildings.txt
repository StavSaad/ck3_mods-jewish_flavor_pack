# Mishpacha-only branches of the vanilla estate building chains, like the Monastery branch of the temple.
# Each branch forks from a vanilla tier, so it is a choice in place of the vanilla upgrade, not an extra slot.
# The Counting House is a full line of its own, replacing the market.
# Building modifiers stack with the earlier tiers of the same chain; parameters do not, so they are repeated per tier.

# Yeshiva sub-branch of the library, beside the Observatory and the Tutor's Office
jip_yeshiva_03 = {
	construction_time = 730
	allowed_domicile_types = { estate }
	slot_type = internal
	previous_building = library_02

	can_construct_potential = {
		jip_can_build_jewish_estate_building_trigger = yes
	}
	can_construct = {
		domicile ?= { has_domicile_building_or_higher = estate_main_02 }
	}

	cost = {
		gold = {
			value = estate_internal_building_normal_cost_tier_3_value
			if = {
				limit = { has_character_modifier = raid_insight_library }
				multiply = 0.75
				desc = raid_insight_library
			}
		}
	}

	parameters = {
		estate_increase_physician_aptitude_3 = yes
		estate_increased_tutor_aptitude_2 = yes
		jip_estate_yeshiva_1 = yes
	}

	character_modifier = {
		monthly_learning_lifestyle_xp_gain_mult = 0.05
		domicile_monthly_piety_add = 0.2
	}

	ai_value = {
		value = 15
		add = estate_building_ai_modifier_value
	}

	asset = {
		icon = "gfx/interface/icons/domicile_building/jip_domicile_building_yeshiva.dds"
		texture = "gfx/interface/window_domiciles/estate_building_villa_byzantine.dds"
		intersectionmask_texture = "gfx/interface/window_domiciles/estate_building_villa_byzantine_mask.png"
	}
}
