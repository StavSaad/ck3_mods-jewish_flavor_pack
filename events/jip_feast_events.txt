namespace = jip_feast

# The mod's festivals, feast variants added by tools/make_feast_override.py (0.15 plan, sections 7 and 9). The meal
# phase starts each chain for the host (jip_feast_meal_phase_effect), the toast phase ends it for everyone
# (jip_feast_toast_phase_effect), and each variant has a guest pulse event (common/on_action/jip_feast_on_actions.txt).
# The year's omen is var:jip_feast_omen on the host (flag:good, flag:bad, else neutral), turned into a one-year
# modifier by jip_feast_omen_modifier_effect
#
# 1000s Akitu (Mesopotamian pagan faith)
#  1001 Host: purification and the Enuma Elish
#  1002 Host: the king's humiliation (the omen)
#  1003 Host: the procession of the gods
#  1004 Host: the fates are fixed (summary)
#  1010 Guest: farewell
#  1020 A ruler who let Akitu lapse: the priests' warning
#  1101 Guest pulse: a priest of Nabu reads your fate
# 2000s Feast of Nisan (Sabians, Sinism)
#  2001 Host: the Moon's fast ends
#  2002 Host: the sacrifice to Sin and the seven (the omen)
#  2004 Host: the mystery of the North (summary)
#  2010 Guest: farewell
#  2101 Guest pulse: an astrologer casts your horoscope
# 3000s Parwanaya (Mandaean rite)
#  3001 Host: white robes and living water
#  3002 Host: the meal for the souls
#  3004 Host: the five days end (summary)
#  3010 Guest: farewell
#  3101 Guest pulse: invited into the river
# 4000s Dehwa Rabba (Mandaean rite)
#  4001 Host: the night of seclusion (the omen)
#  4002 Host: the New Year greeted
#  4004 Host: the year begins (summary)
#  4010 Guest: farewell
#  4101 Guest pulse: tales in the night
# 5000s Pesach (every Jewish faith; 0.16 plan, section 1)
#  5001 Host: Kiddush and the four cups
#  5002 Host: the questions (the youngest child, scope:jip_seder_child, or the host)
#  5003 Host: the afikoman ransomed (Haggadah faiths, with a child)
#  5004 Host: Elijah's cup and the open door (Haggadah faiths)
#  5006 Host: eaten in haste (Karaites, Samaritans, Beta Israel)
#  5005 Host: "Next year in Jerusalem" (summary; the Exilarch's seder, the Korban Pesach, Gerizim)
#  5010 Guest: farewell
#  5020 The household seder (yearly in April, for Jewish playable characters who sat at no grand seder)
#  5101 Guest pulse: Dayenu

# Purification and the Enuma Elish
jip_feast.1001 = {
	type = activity_event
	title = jip_feast.1001.t
	desc = jip_feast.1001.desc
	theme = feast_activity
	widgets = {
		widget = {
			gui = "jip_event_source_badge"
			container = "foreground_shader_vfx_container"
		}
	}
	left_portrait = {
		character = root
		animation = prayer
	}
	right_portrait = {
		trigger = { exists = scope:jip_feast_priest }
		character = scope:jip_feast_priest
		animation = personality_zealous
	}

	immediate = {
		jip_feast_save_scopes_effect = yes
	}

	# Listen to every line
	option = {
		name = jip_feast.1001.a
		add_piety = minor_piety_gain
		if = {
			limit = { has_trait = zealous }
			add_stress = minor_stress_loss
		}
		ai_chance = {
			base = 50
			modifier = {
				add = 50
				has_trait = zealous
			}
		}
	}

	# Have it recited before the guests
	option = {
		name = jip_feast.1001.b
		add_prestige = minor_prestige_gain
		ai_chance = {
			base = 40
			modifier = {
				add = 40
				has_trait = arrogant
			}
		}
	}

	# Read the tablets yourself
	option = {
		name = jip_feast.1001.c
		trigger = { learning >= 12 }
		add_learning_lifestyle_xp = medium_lifestyle_xp
		add_piety = miniscule_piety_gain
		ai_chance = {
			base = 30
			modifier = {
				add = 40
				has_trait = scholar
			}
		}
	}

	after = {
		trigger_event = {
			id = jip_feast.1002
			days = 3
		}
	}
}

# The king's humiliation: the high priest strikes the king, and if tears come the god is pleased
jip_feast.1002 = {
	type = activity_event
	title = jip_feast.1002.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:jip_feast_priest }
				desc = jip_feast.1002.desc_priest
			}
			desc = jip_feast.1002.desc
		}
		desc = jip_feast.1002.desc_end
	}
	theme = feast_activity
	widgets = {
		widget = {
			gui = "jip_event_source_badge"
			container = "foreground_shader_vfx_container"
		}
	}
	left_portrait = {
		character = root
		animation = pain
	}
	right_portrait = {
		trigger = { exists = scope:jip_feast_priest }
		character = scope:jip_feast_priest
		animation = disapproval
	}

	# Take the blow
	option = {
		name = jip_feast.1002.a
		custom_tooltip = jip_feast.1002.omen_tt
		hidden_effect = {
			random_list = {
				50 = {
					modifier = {
						add = 25
						has_trait = humble
					}
					modifier = {
						add = 15
						has_trait = zealous
					}
					set_variable = {
						name = jip_feast_omen
						value = flag:good
					}
				}
				50 = {
					modifier = {
						add = 25
						has_trait = arrogant
					}
					set_variable = {
						name = jip_feast_omen
						value = flag:bad
					}
				}
			}
		}
		ai_chance = {
			base = 60
		}
	}

	# Let the tears come
	option = {
		name = jip_feast.1002.b
		trigger = {
			OR = {
				has_trait = humble
				has_trait = zealous
				has_trait = compassionate
			}
		}
		add_stress = minor_stress_gain
		custom_tooltip = jip_feast.1002.omen_likely_tt
		hidden_effect = {
			random_list = {
				85 = {
					set_variable = {
						name = jip_feast_omen
						value = flag:good
					}
				}
				15 = {
					set_variable = {
						name = jip_feast_omen
						value = flag:bad
					}
				}
			}
		}
		ai_chance = {
			base = 80
		}
	}

	# No priest strikes a king
	option = {
		name = jip_feast.1002.c
		add_prestige = minor_prestige_gain
		add_piety = medium_piety_loss
		custom_tooltip = jip_feast.1002.omen_bad_tt
		hidden_effect = {
			set_variable = {
				name = jip_feast_omen
				value = flag:bad
			}
		}
		ai_chance = {
			base = 5
			modifier = {
				add = 60
				has_trait = arrogant
			}
			modifier = {
				add = 30
				has_trait = wrathful
			}
		}
	}

	after = {
		trigger_event = {
			id = jip_feast.1003
			days = 3
		}
	}
}

# "Next year in Jerusalem" (host summary, toast phase)
jip_feast.5005 = {
	type = activity_event
	title = jip_feast.5005.t
	desc = {
		desc = jip_feast.5005.desc
		triggered_desc = {
			trigger = { jip_temple_stands_trigger = yes }
			desc = jip_feast.5005.desc_temple
		}
		triggered_desc = {
			trigger = {
				faith = faith:samaritan
				jip_feast_at_holy_site_trigger = yes
			}
			desc = jip_feast.5005.desc_gerizim
		}
		triggered_desc = {
			trigger = { jip_is_exilarch_trigger = yes }
			desc = jip_feast.5005.desc_exilarch
		}
		first_valid = {
			triggered_desc = {
				trigger = { jip_pesach_host_holds_jerusalem_trigger = yes }
				desc = jip_feast.5005.desc_this_year
			}
			desc = jip_feast.5005.desc_next_year
		}
	}
	theme = feast_activity
	widgets = {
		widget = {
			gui = "jip_event_source_badge"
			container = "foreground_shader_vfx_container"
		}
	}
	left_portrait = {
		character = root
		animation = personality_zealous
	}

	option = {
		name = {
			trigger = { jip_pesach_host_holds_jerusalem_trigger = yes }
			text = jip_feast.5005.a_this_year
		}
		name = {
			trigger = { NOT = { jip_pesach_host_holds_jerusalem_trigger = yes } }
			text = jip_feast.5005.a
		}
		add_piety = medium_piety_gain
		if = {
			limit = { jip_pesach_host_holds_jerusalem_trigger = yes }
			add_prestige = medium_prestige_gain
			add_legitimacy = minor_legitimacy_gain
			add_character_modifier = {
				modifier = jip_pesach_this_year_in_jerusalem
				years = 1
			}
		}
		else = {
			add_character_modifier = {
				modifier = jip_pesach_next_year_in_jerusalem
				years = 1
			}
		}
		# The Korban Pesach at the Temple
		if = {
			limit = { jip_temple_stands_trigger = yes }
			add_piety = medium_piety_gain
		}
		# The Samaritans' lamb on Mount Gerizim
		if = {
			limit = {
				faith = faith:samaritan
				jip_feast_at_holy_site_trigger = yes
			}
			add_piety = medium_piety_gain
		}
		# The Exilarch's seder, with the heads of the academies
		if = {
			limit = { jip_is_exilarch_trigger = yes }
			add_legitimacy = medium_legitimacy_gain
		}
	}
}

# The household seder
jip_feast.5020 = {
	type = character_event
	title = jip_feast.5020.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { jip_pesach_paschal_lamb_trigger = yes }
				desc = jip_feast.5020.desc_lamb
			}
			triggered_desc = {
				trigger = { faith = faith:karaism }
				desc = jip_feast.5020.desc_karaite
			}
			desc = jip_feast.5020.desc
		}
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:jip_seder_child }
				desc = jip_feast.5020.desc_child
			}
			desc = jip_feast.5020.desc_no_child
		}
	}
	theme = family
	widgets = {
		widget = {
			gui = "jip_event_source_badge"
			container = "foreground_shader_vfx_container"
		}
	}
	override_background = { reference = bp1_relaxing_room_mena }
	left_portrait = {
		character = root
		animation = happiness
	}
	right_portrait = {
		trigger = { exists = scope:jip_seder_child }
		character = scope:jip_seder_child
		animation = interested
	}

	trigger = {
		is_alive = yes
		religion = religion:judaism_religion
		is_imprisoned = no
		NOT = { has_character_flag = jip_kept_pesach }
	}

	immediate = {
		jip_pesach_find_child_effect = yes
	}

	# Gather the household
	option = {
		name = jip_feast.5020.a
		add_piety = minor_piety_gain
		add_stress = minor_stress_loss
		every_close_family_member = {
			limit = {
				is_alive = yes
				is_courtier_of = root
			}
			add_opinion = {
				target = root
				modifier = pleased_opinion
				opinion = 10
			}
		}
		ai_chance = {
			base = 50
		}
	}

	# Let the youngest ask
	option = {
		name = jip_feast.5020.b
		trigger = { exists = scope:jip_seder_child }
		add_piety = minor_piety_gain
		scope:jip_seder_child = {
			add_opinion = {
				target = root
				modifier = pleased_opinion
				opinion = 15
			}
			random = {
				chance = 30
				add_learning_skill = 1
			}
		}
		ai_chance = {
			base = 50
		}
	}

	# A quiet seder, and early to bed
	option = {
		name = jip_feast.5020.c
		trigger = {
			OR = {
				has_trait = shy
				has_trait = lazy
				has_trait = cynical
			}
		}
		add_stress = medium_stress_loss
		ai_chance = {
			base = 40
		}
	}
}
