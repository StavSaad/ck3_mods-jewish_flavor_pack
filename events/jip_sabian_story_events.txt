# The Return of the Moon God, the Sabian story (common/story_cycles/jip_story_sabian_revival.txt; 0.15 plan, section 10).
# Root = the story owner, the head of Thabit ibn Qurra's house.
#  1001 The Sabians of Baghdad (the story begins, players)
#  1002 The Moon God's children (the story offered to any other Sabian player at the start)
#  1010 A patron of the sciences (once)
#  1020 The elders of Harran (once)
#  1030 A disputation on the Sabians (every 8 years at most)
#  1040 An initiate's secret reaches a Muslim (scope:jip_initiate)
#  1050 The stars over the palace (every 5 years at most)
#  1090 The road to Harran (standing reaches 10: Act 1 ends)
#  1100 The mysteries (Keep the Mysteries; scope:jip_initiate)
# Learned answers succeed with jip_sabian_story_learning_chance_value.

# A patron of the sciences
jip_sabian_story.1010 = {
	type = character_event
	title = jip_sabian_story.1010.t
	desc = jip_sabian_story.1010.desc
	theme = learning
	widgets = {
		widget = {
			gui = "jip_event_source_badge"
			container = "foreground_shader_vfx_container"
		}
	}
	override_background = { reference = study }
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:jip_patron
		animation = personality_rational
	}

	trigger = {
		jip_sabian_story_has_muslim_patron_trigger = yes
	}

	immediate = {
		random_owned_story = {
			type = story_jip_sabian_revival
			var:jip_sabian_patron = { save_scope_as = jip_patron }
		}
		ordered_courtier = {
			limit = {
				house ?= root.house
				age >= 14
			}
			order_by = learning
			save_scope_as = jip_pupil
		}
	}

	# Dispute before the court
	option = {
		name = jip_sabian_story.1010.a
		random_list = {
			0 = {
				desc = jip_sabian_story.1010.a.success
				modifier = {
					add = jip_sabian_story_learning_chance_value
				}
				jip_sabian_story_raise_standing_effect = { VALUE = 2 }
				add_prestige = medium_prestige_gain
				scope:jip_patron = {
					add_opinion = {
						target = root
						modifier = impressed_opinion
						opinion = 20
					}
				}
			}
			0 = {
				desc = jip_sabian_story.1010.a.failure
				modifier = {
					add = jip_sabian_story_learning_failure_value
				}
				jip_sabian_story_lower_standing_effect = { VALUE = 1 }
				add_prestige = minor_prestige_loss
			}
		}
		ai_chance = {
			base = 50
			modifier = {
				add = learning
			}
		}
	}

	# Send a young scholar of the house
	option = {
		name = jip_sabian_story.1010.b
		trigger = { exists = scope:jip_pupil }
		scope:jip_pupil = {
			add_learning_lifestyle_xp = medium_lifestyle_xp
		}
		jip_sabian_story_raise_standing_effect = { VALUE = 1 }
		ai_chance = {
			base = 30
		}
	}

	# Decline
	option = {
		name = jip_sabian_story.1010.c
		add_learning_lifestyle_xp = minor_lifestyle_xp
		ai_chance = {
			base = 5
		}
	}
}

# The Sabian asks for Harran (root = the patron)
jip_sabian_story.2010 = {
	type = character_event
	title = jip_sabian_story.2010.t
	desc = jip_sabian_story.2010.desc
	theme = realm
	widgets = {
		widget = {
			gui = "jip_event_source_badge"
			container = "foreground_shader_vfx_container"
		}
	}
	override_background = { reference = throne_room_mediterranean }
	left_portrait = {
		character = scope:jip_petitioner
		animation = personality_rational
	}
	right_portrait = {
		character = scope:jip_harran_holder
		trigger = { scope:jip_harran_holder != root }
		animation = worry
	}

	trigger = {
		exists = scope:jip_petitioner
		scope:jip_petitioner = { jip_sabian_can_seek_harran_trigger = yes }
	}

	immediate = {
		title:c_harran.holder = { save_scope_as = jip_harran_holder }
	}

	# Grant it
	option = {
		name = jip_sabian_story.2010.a
		save_scope_as = jip_harran_liege
		if = {
			limit = { scope:jip_harran_holder != root }
			scope:jip_harran_holder = {
				add_opinion = {
					target = root
					modifier = angry_opinion
					opinion = -30
				}
			}
		}
		scope:jip_petitioner = {
			jip_sabian_take_harran_effect = { TYPE = granted }
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 30
			}
			trigger_event = jip_sabian_story.2011
		}
		ai_chance = {
			base = 25
			opinion_modifier = {
				opinion_target = scope:jip_petitioner
				multiplier = 0.75
			}
			modifier = {
				add = 20
				primary_title = { has_variable = jip_sabians_recognised }
			}
			modifier = {
				add = 15
				employs_court_position = jip_sabian_astrologer_court_position
			}
			modifier = {
				add = -30
				scope:jip_harran_holder = root
			}
			modifier = {
				add = -15
				scope:jip_harran_holder != root
				scope:jip_harran_holder = {
					opinion = {
						target = root
						value >= 30
					}
				}
			}
			ai_value_modifier = {
				ai_zeal = -0.5
			}
			min = 0
		}
	}

	# Refuse
	option = {
		name = jip_sabian_story.2010.b
		scope:jip_petitioner = {
			trigger_event = jip_sabian_story.2012
		}
		ai_chance = {
			base = 50
		}
	}
}

	type = character_event

			gui = "jip_event_source_badge"

		}

		character = root
		animation = anger
	}

		animation = personality_bold
	}

		exists = scope:jip_rebel
		scope:jip_rebel = {
			jip_sabian_can_proclaim_freedom_trigger = yes
			liege = root
		}
	}

	# Let Harran go
	option = {
		name = jip_sabian_story.3010.a
		jip_sabian_grant_freedom_effect = yes
		add_prestige = medium_prestige_loss

			trigger_event = jip_sabian_story.3012
		}
		ai_chance = {
			base = 5
			modifier = {
				add = 40
				is_at_war = yes
			}
			modifier = {
				add = 20
				opinion = {
					target = scope:jip_rebel
					value >= 50
				}
			}
			modifier = {
				add = 15
				primary_title = { has_variable = jip_sabians_recognised }
			}
			ai_value_modifier = {
				ai_boldness = -0.5
			}
			min = 0
		}
	}

	option = {
		name = jip_sabian_story.3010.b
		scope:jip_rebel = {
			trigger_event = jip_sabian_story.3011
		}
		ai_chance = {
			base = 60
		}
	}
}
