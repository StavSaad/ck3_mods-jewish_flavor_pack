namespace = jip_sabian

# The Sabians of Harran and the return of the old gods (0.15 plan, sections 2, 4 and 10)
#  0001 The House of Rejoicing restored (the owner of jip_gp_temple_of_sin, on completion): proclaim the old gods, or
#       remain a Sabian. The owner of the Sabian story who hasn't unveiled the mysteries does it here instead
#       (jip_sabian_unveil_the_mysteries_effect, then jip_sabian_story.4010)
#  0100 The Question of the Sabians (a Sabian or Mandaean vassal or family, from jip_question_the_sabians_decision): pay,
#       defend the claim to be a people of the Book, or convert
#  0200 A Sabian scholar (a Muslim duke or above, from jip_sabian_scholar_on_action): commission translations, set them to
#       the realm's accounts, or send them away

# The House of Rejoicing restored
# Root = the project's owner (scope:jip_temple_builder)
jip_sabian.0001 = {
	type = character_event
	title = jip_sabian.0001.t
	desc = jip_sabian.0001.desc
	theme = faith
	widgets = {
		widget = {
			gui = "jip_event_source_badge"
			container = "foreground_shader_vfx_container"
		}
	}
	override_background = { reference = temple }
	left_portrait = {
		character = root
		animation = personality_zealous
	}
	right_portrait = {
		trigger = { exists = scope:jip_sabian_priest }
		character = scope:jip_sabian_priest
		animation = prayer
	}

	immediate = {
		cp:councillor_court_chaplain ?= { save_scope_as = jip_sabian_priest }
	}

	# Proclaim the old gods
	option = {
		name = jip_sabian.0001.a
		trigger = {
			NOT = { faith = faith:mesopotamian_pagan }
			jip_sabian_story_awaits_unveiling_trigger = no
		}
		jip_proclaim_old_gods_effect = yes
		add_piety = major_piety_gain
		ai_chance = {
			base = 50
			modifier = {
				add = 50
				has_trait = zealous
			}
			modifier = {
				add = 25
				learning >= 12
			}
		}
	}

	# Remain a Sabian: the temple serves the Moon all the same
	option = {
		name = jip_sabian.0001.b
		trigger = {
			rite = rite:sabianism
			jip_sabian_story_awaits_unveiling_trigger = no
		}
		add_piety = medium_piety_gain
		add_prestige = medium_prestige_gain
		ai_chance = {
			base = 50
			modifier = {
				add = 50
				has_trait = content
			}
		}
	}

	# The Sabian story's owner: the mysteries are unveiled
	option = {
		name = jip_sabian.0001.d
		trigger = { jip_sabian_story_awaits_unveiling_trigger = yes }
		custom_tooltip = jip_sabian_unveil_the_mysteries_effect_tt
		jip_sabian_unveil_the_mysteries_effect = yes
		add_piety = major_piety_gain
		trigger_event = jip_sabian_story.4010
	}

	# Already of the old faith
	option = {
		name = jip_sabian.0001.c
		trigger = {
			faith = faith:mesopotamian_pagan
			jip_sabian_story_awaits_unveiling_trigger = no
		}
		add_piety = major_piety_gain
	}
}

# The Question of the Sabians: the questioner asks the Sabians and Mandaeans of the realm to prove themselves a people of
# the Book. A defence that convinces them wins a recognition (var:jip_sabians_recognised on their primary title).
# Root = the Sabian or Mandaean (a landed vassal, or the head of a family in the realm), scope:jip_questioner = the ruler
jip_sabian.0100 = {
	type = character_event
	title = jip_sabian.0100.t
	desc = {
		desc = jip_sabian.0100.desc
		first_valid = {
			triggered_desc = {
				trigger = { rite = rite:mandeaism }
				desc = jip_sabian.0100.desc_mandaean
			}
			desc = jip_sabian.0100.desc_sabian
		}
	}
	theme = faith
	widgets = {
		widget = {
			gui = "jip_event_source_badge"
			container = "foreground_shader_vfx_container"
		}
	}
	override_background = { reference = throne_room }
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:jip_questioner
		animation = dismissal
	}

	trigger = {
		exists = scope:jip_questioner
		scope:jip_questioner = { is_alive = yes }
		jip_is_sabian_or_mandaean_trigger = yes
	}

	# Pay for the protection
	option = {
		name = jip_sabian.0100.a
		trigger = { gold >= jip_sabian_question_fee_value }
		show_as_unavailable = { always = yes }
		remove_short_term_gold = jip_sabian_question_fee_value
		scope:jip_questioner = { add_gold = jip_sabian_question_fee_value }
		scope:jip_questioner = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = jip_sabian.0100.a.toast
				left_icon = root
			}
		}
		ai_chance = {
			base = 40
			modifier = {
				add = 20
				gold >= 300
			}
			ai_value_modifier = {
				ai_greed = -0.5
			}
		}
	}

	# Defend the claim
	option = {
		name = jip_sabian.0100.b
		random_list = {
			0 = {
				desc = jip_sabian.0100.b.success
				modifier = {
					add = jip_sabian_defence_chance_value
				}
				add_prestige = medium_prestige_gain
				add_piety = medium_piety_gain
				jip_sabian_story_raise_standing_effect = { VALUE = 2 }
				scope:jip_questioner = {
					primary_title = {
						set_variable = {
							name = jip_sabians_recognised
							value = yes
							years = jip_sabians_recognised_years_value
						}
					}
				}
				custom_tooltip = jip_sabian.0100.b.success_tt
				scope:jip_questioner = {
					send_interface_toast = {
						type = event_toast_effect_neutral
						title = jip_sabian.0100.b.success_toast
						left_icon = root
					}
				}
			}
			0 = {
				desc = jip_sabian.0100.b.failure
				modifier = {
					add = {
						value = 100
						subtract = jip_sabian_defence_chance_value
					}
				}
				remove_short_term_gold = jip_sabian_question_failed_fee_value
				add_prestige = medium_prestige_loss
				jip_sabian_story_lower_standing_effect = { VALUE = 2 }
				scope:jip_questioner = {
					add_gold = jip_sabian_question_failed_fee_value
					add_piety = minor_piety_gain
				}
				scope:jip_questioner = {
					send_interface_toast = {
						type = event_toast_effect_neutral
						title = jip_sabian.0100.b.failure_toast
						left_icon = root
					}
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = {
				add = learning
			}
			ai_value_modifier = {
				ai_zeal = 0.5
				ai_boldness = 0.25
			}
		}
	}

	# Convert
	option = {
		name = jip_sabian.0100.c
		if = {
			limit = { government_has_flag = government_is_familial }
			custom_tooltip = jip_sabian.0100.c.family_tt
		}
		set_character_faith = scope:jip_questioner.faith
		scope:jip_questioner = { add_piety = medium_piety_gain }
		scope:jip_questioner = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = jip_sabian.0100.c.toast
				left_icon = root
			}
		}
		ai_chance = {
			base = 5
			modifier = {
				add = 20
				has_trait = cynical
			}
			modifier = {
				factor = 0
				has_trait = zealous
			}
			ai_value_modifier = {
				ai_zeal = -0.5
			}
		}
	}
}

# A Sabian scholar, a courtier or the head of a Sabian family in the realm, offers their learning: Greek books put into
# Arabic, or the realm's sums. Patronage raises the Sabians' standing when the scholar carries the Sabian story.
# Root = the Muslim ruler; scope:jip_sabian_scholar = the scholar (the story's owner first)
jip_sabian.0200 = {
	type = character_event
	title = jip_sabian.0200.t
	desc = {
		desc = jip_sabian.0200.desc
		first_valid = {
			triggered_desc = {
				trigger = { scope:jip_sabian_scholar = { is_ruler = yes } }
				desc = jip_sabian.0200.desc_family
			}
			desc = jip_sabian.0200.desc_courtier
		}
	}
	theme = learning
	widgets = {
		widget = {
			gui = "jip_event_source_badge"
			container = "foreground_shader_vfx_container"
		}
	}
	override_background = { reference = study }
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:jip_sabian_scholar
		animation = scheme
	}

	trigger = {
		jip_can_patronise_sabian_scholar_trigger = yes
	}

	immediate = {
		set_variable = {
			name = jip_sabian_scholar_cooldown
			years = 8
		}
		every_courtier = {
			limit = { jip_sabian_scholar_for_trigger = { RULER = root } }
			add_to_temporary_list = jip_sabian_scholars
		}
		every_ruler = {
			limit = { jip_sabian_scholar_for_trigger = { RULER = root } }
			add_to_temporary_list = jip_sabian_scholars
		}
		random_in_list = {
			list = jip_sabian_scholars
			weight = {
				base = 1
				modifier = {
					add = 20
					jip_owns_sabian_story_trigger = yes
				}
				modifier = {
					add = learning
				}
			}
			save_scope_as = jip_sabian_scholar
		}
	}

	# Commission translations
	option = {
		name = jip_sabian.0200.a
		trigger = { gold >= medium_gold_value }
		show_as_unavailable = { always = yes }
		remove_short_term_gold = medium_gold_value
		add_learning_lifestyle_xp = medium_lifestyle_xp
		add_prestige = minor_prestige_gain
		scope:jip_sabian_scholar = {
			add_gold = medium_gold_value
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 20
			}
			jip_sabian_story_raise_standing_effect = { VALUE = 1 }
		}
	}

	# Set them to the realm's sums
	option = {
		name = jip_sabian.0200.b
		add_stewardship_lifestyle_xp = medium_lifestyle_xp
		capital_county ?= {
			change_development_progress = medium_development_progress_gain
		}
		scope:jip_sabian_scholar = {
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = 10
			}
			jip_sabian_story_raise_standing_effect = { VALUE = 1 }
		}
	}

	# Send them away
	option = {
		name = jip_sabian.0200.c
		add_piety = minor_piety_gain
		scope:jip_sabian_scholar = {
			add_opinion = {
				target = root
				modifier = insulted_opinion
				opinion = -15
			}
			jip_sabian_story_lower_standing_effect = { VALUE = 1 }
		}
	}
}
