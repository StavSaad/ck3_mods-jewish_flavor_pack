namespace = jip_palmyra

# Greater Aram is founded (create_e_aram_decision)
jip_palmyra.0001 = {
	type = character_event
	title = jip_palmyra.0001.t
	desc = jip_palmyra.0001.desc
	theme = realm
	widgets = {
		widget = {
			gui = "jip_event_source_badge"
			container = "foreground_shader_vfx_container"
		}
	}
	left_portrait = {
		character = root
		animation = ecstasy
	}

	immediate = {
		play_music_cue = "mx_cue_positive_effect"
		create_aram_decision_effect = yes
	}

	# Rule from Palmyra, as Odaenathus and Zenobia did
	option = {
		name = jip_palmyra.0001.a
		trigger = {
			NOT = { capital_county = title:c_palmyra }
		}
		set_realm_capital = title:c_palmyra
		stress_impact = {
			ambitious = major_stress_loss
		}
	}

	# Keep the court where it is
	option = {
		name = jip_palmyra.0001.b
		title:e_aram = {
			set_capital_county = root.capital_county
		}
		stress_impact = {
			ambitious = medium_stress_loss
		}
	}

	after = {
		# The founder may bring back Palmyra's gods
		if = {
			limit = {
				NOT = { faith = faith:aramean_pagan }
			}
			trigger_event = jip_palmyra.0002
		}
	}
}

# The gods of Tadmor: the founder of Greater Aram may restore the gods of Aram (0.15 plan, section 14)
jip_palmyra.0002 = {
	type = character_event
	title = jip_palmyra.0002.t
	desc = jip_palmyra.0002.desc
	theme = faith
	widgets = {
		widget = {
			gui = "jip_event_source_badge"
			container = "foreground_shader_vfx_container"
		}
	}
	left_portrait = {
		character = root
		animation = thinking
	}

	# Bring back Bel and the gods of Aram
	option = {
		name = jip_palmyra.0002.a
		add_character_flag = jip_restored_gods_of_aram
		jip_restore_gods_of_aram_effect = { RITE = belism }
		stress_impact = {
			zealous = major_stress_impact_gain
			cynical = minor_stress_impact_loss
		}
		ai_chance = {
			base = 10
		}
	}

	# Keep the faith
	option = {
		name = jip_palmyra.0002.b
		stress_impact = {
			cynical = minor_stress_impact_gain
		}
		ai_chance = {
			base = 90
		}
	}
}
