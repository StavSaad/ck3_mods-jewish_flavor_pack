scripted_trigger jip_hebrew_0002_is_in_israel_trigger = {
	capital_province = {
		county = {
			any_this_title_or_de_jure_above = {
				this = title:k_israel
			}
		}
	}
}

#Someone reformed Hebrew culture!
jip_hebrew.0002 = {
	type = character_event
	title = jip_hebrew.0001.t
	desc = {
		desc = jip_hebrew.0002.start.desc
		first_valid = {
			triggered_desc = {
				trigger = {
					has_RelationToMe_relation = { CHARACTER = scope:hebrew_former }
				}
				desc = jip_hebrew.0002.relation_former.desc
			}
			desc = jip_hebrew.0002.former.desc
		}	
		desc = jip_hebrew.0002.end.desc		
	}
	theme = culture_change
	left_portrait = {
		character = scope:hebrew_former
		animation = admiration
	}

	immediate = {
		#If in israel, play the appropriate sound effect.
		if = {
			limit = { jip_hebrew_0002_is_in_israel_trigger = yes }
			play_music_cue = "mx_cue_positive_effect"
		}
	}

	#OK
	option = {
		trigger = { jip_hebrew_0002_is_in_israel_trigger = no }
		name = {
			trigger = { faith = scope:hebrew_former.faith }
			text = jip_hebrew.0002.a1
		}
		name = {
			trigger = { NOT = { faith = scope:hebrew_former.faith } }
			text = name_i_see
		}
	}

	#Convert to Hebrew
	option = {
		trigger = { jip_hebrew_0002_is_in_israel_trigger = yes }
		name = jip_hebrew.0002.b

		every_courtier = {
			limit = { has_same_culture_as = root }
			add_to_list = convert_list
		}

		set_culture = culture:hebrew

		every_in_list = {
			list = convert_list
			custom = hebrew_convert_list_court
			set_culture = culture:hebrew
		}

		ai_chance = {
			base = 1
		}
	}

	#Keep my culture
	option = {
		trigger = { jip_hebrew_0002_is_in_israel_trigger = yes }
		name = jip_hebrew.0002.c

		reverse_add_opinion = {
			target = scope:hebrew_former
			modifier = disappointed_opinion
			opinion = -15
		}

		ai_chance = {
			base = 0
		}
	}
}

# Restore HoF Event after the third temple is built
jip_hebrew.0040 = {
	type = character_event
	title = jip_hebrew.0040.t #change
	desc = {
		desc = jip_hebrew.0040.desc #change
	}
	theme = faith
	left_portrait = {
		character = root
		animation = personality_zealous
	}

	immediate = {
		custom_description_no_bullet = {
			text = judaism_hof_restored_effect
			faith:rabbinism = {
				remove_doctrine = special_doctrine_not_allowed_to_hof
			}
			faith:karaism = {
				remove_doctrine = special_doctrine_not_allowed_to_hof
			}
			faith:kabarism = {
				remove_doctrine = special_doctrine_not_allowed_to_hof
			}
			faith:malabarism = {
				remove_doctrine = special_doctrine_not_allowed_to_hof
			}
			faith:haymanot = {
				remove_doctrine = special_doctrine_not_allowed_to_hof
			}
			faith:kohanism = {
				remove_doctrine = special_doctrine_not_allowed_to_hof
			}
		}
	}

	option = {
		name = jip_hebrew.0040.a # NOTHING for non-samaritans
		trigger = {
			NOT = {
				faith = faith:samaritan
			}
		}
		stress_impact = {
			zealous = medium_stress_loss
			diligent = medium_stress_loss
		}
	}

	option = {
		name = jip_hebrew.0040.b # Restore Kohen Gadol for Judaism
		trigger = {
			OR = {
				faith = faith:rabbinism
				faith = faith:karaism
				faith = faith:kabarism
				faith = faith:malabarism
				faith = faith:haymanot
				faith = faith:kohanism
			}
		}
		faith:rabbinism = {
			remove_doctrine = doctrine_no_head
			add_doctrine = doctrine_spiritual_head
			set_religious_head_title = faith:kohanism.religious_head_title
		}
		faith = {
			save_scope_as = faith
		}
		create_head_of_faith_title_effect = yes
		stress_impact = { 
			zealous = medium_stress_gain
			diligent = medium_stress_loss
		}
	}

	option = {
		name = jip_hebrew.0040.c # Convert to Kohenism
		trigger = {
			AND = {
				NOT = {
					faith = faith:kohanism
				}
				NOT = {
					faith = faith:samaritan
				}
			}
		}
		set_character_faith = faith:kohanism
		faith:kohanism = {
			save_scope_as = faith
		}
		create_head_of_faith_title_effect = yes
		create_title_and_vassal_change = {
			type = swear_fealty
			save_scope_as = change
		}
		title:d_kohen_gadol = {
			holder = {
				change_liege = {
					liege = root
					change = scope:change
				}
				resolve_title_and_vassal_change = scope:change
			}
		}
		stress_impact = {
			zealous = major_stress_gain
			ambitious = major_stress_loss
			diligent = medium_stress_loss
		}
	}

	option = {
		name = jip_hebrew.0040.d # nothing for Samaritans
		trigger = {
			faith = faith:samaritan
		}
	}
}
